func (r p07Report) f(format string, args ...any) { fmt.Fprintf(r.out, format+"\n", args...) }

func p07Err(err error) string {
	if err == nil {
		return "<nil>"
	}
	var app *apperror.AppError
	if errors.As(err, &app) {
		return fmt.Sprintf("AppError{code=%q type=%q status=%d detail=%q}", app.Code, app.Type, app.Status, app.Detail)
	}
	return fmt.Sprintf("%s: %v", resource.ErrorCode(err), err)
}

func p07Marshal(v any) ([]byte, error) {
	var buffer bytes.Buffer
	encoder := json.NewEncoder(&buffer)
	encoder.SetEscapeHTML(false)
	if err := encoder.Encode(v); err != nil {
		return nil, err
	}
	return bytes.TrimRight(buffer.Bytes(), "\n"), nil
}

// p07StripTimeNullable removes nullable from the two Kubernetes time components
// of an OpenAPI document and re-encodes it without touching unrelated subtrees.
func p07StripTimeNullable(document []byte) ([]byte, int, error) {
	var top map[string]json.RawMessage
	if err := json.Unmarshal(document, &top); err != nil {
		return nil, 0, err
	}
	var components map[string]json.RawMessage
	if err := json.Unmarshal(top["components"], &components); err != nil {
		return nil, 0, err
	}
	var schemas map[string]json.RawMessage
	if err := json.Unmarshal(components["schemas"], &schemas); err != nil {
		return nil, 0, err
	}
	stripped := 0
	for _, name := range []string{p07TimeComponent, p07MicroTimeComponent} {
		raw, ok := schemas[name]
		if !ok {
			continue
		}
		var fields map[string]json.RawMessage
		if err := json.Unmarshal(raw, &fields); err != nil {
			return nil, 0, err
		}
		if _, present := fields["nullable"]; !present {
			continue
		}
		delete(fields, "nullable")
		encoded, err := p07Marshal(fields)
		if err != nil {
			return nil, 0, err
		}
		schemas[name] = encoded
		stripped++
	}
	encodedSchemas, err := p07Marshal(schemas)
	if err != nil {
		return nil, 0, err
	}
	components["schemas"] = encodedSchemas
	encodedComponents, err := p07Marshal(components)
	if err != nil {
		return nil, 0, err
	}
	top["components"] = encodedComponents
	out, err := p07Marshal(top)
	return out, stripped, err
}

func p07Component(document []byte, name string) (string, bool) {
	var top struct {
		Components struct {
			Schemas map[string]json.RawMessage `json:"schemas"`
		} `json:"components"`
	}
	if err := json.Unmarshal(document, &top); err != nil {
		return "", false
	}
	raw, ok := top.Components.Schemas[name]
	return string(raw), ok
}

func p07HostDigest(dialect string, document []byte) string {
	sum := sha256.Sum256([]byte(p07DocDigestPrefix + dialect + "\x00" + string(document)))
	return hex.EncodeToString(sum[:])
}

func p07SHA(document []byte) string {
	sum := sha256.Sum256(document)
	return hex.EncodeToString(sum[:])
}

func p07StartPlugin(t *testing.T, env p07Env, rep p07Report) *p07Plugin {
	t.Helper()
	home := filepath.Join(env.Work, "home")
	state := filepath.Join(env.Work, "state")
	sockets, err := os.MkdirTemp("/tmp/p07-acc/live", "s-")
	require.NoError(t, err)
	t.Cleanup(func() { _ = os.RemoveAll(sockets) })
	require.NoError(t, os.MkdirAll(filepath.Join(home, ".kube"), 0o700))
	require.NoError(t, os.MkdirAll(filepath.Join(env.Work, "kubectl-home"), 0o700))
	pluginDir := filepath.Join(state, "plugins", p07PluginID)
	require.NoError(t, os.MkdirAll(pluginDir, 0o700))
	manifest, err := os.ReadFile(env.PluginYAML)
	require.NoError(t, err)
	require.NoError(t, os.WriteFile(filepath.Join(pluginDir, "plugin.yaml"), manifest, 0o600))
	kubeconfig, err := os.ReadFile(env.Kubeconfig)
	require.NoError(t, err)
	homeKubeconfig := filepath.Join(home, ".kube", "config")
	require.NoError(t, os.WriteFile(homeKubeconfig, kubeconfig, 0o600))
	for _, context := range []string{"p07-a", "p07-b"} {
		command := exec.Command("kubectl", "--kubeconfig", homeKubeconfig, "config", "set-context", context, "--cluster="+p07MainConn, "--user="+p07MainConn)
		command.Env = []string{"PATH=/usr/bin:/bin:/opt/homebrew/bin:/usr/local/bin", "HOME=" + filepath.Join(env.Work, "kubectl-home")}
		output, err := command.CombinedOutput()
		require.NoError(t, err, string(output))
	}

	meta := config.PluginMeta{ID: p07PluginID}
	handshake := meta.GenerateHandshakeConfig()
	command := exec.Command(env.Binary)
	command.Env = []string{
		"HOME=" + home, "USERPROFILE=" + home, "TMPDIR=" + sockets, "TMP=" + sockets, "TEMP=" + sockets,
		"PATH=/usr/bin:/bin:/opt/homebrew/bin:/usr/local/bin",
		"XDG_CONFIG_HOME=" + filepath.Join(home, "config"), "XDG_CACHE_HOME=" + filepath.Join(home, "cache"), "XDG_DATA_HOME=" + filepath.Join(home, "data"),
		"OMNIVIEW_STATE_DIR=" + state, "OMNIVIEW_TELEMETRY_ENABLED=false", "OMNIVIEW_TELEMETRY_PROFILING=false",
		handshake.MagicCookieKey + "=" + handshake.MagicCookieValue, "PLUGIN_PROTOCOL_VERSIONS=1",
	}
	logFile, err := os.Create(filepath.Join(env.Work, "plugin-child.log"))
	require.NoError(t, err)
	t.Cleanup(func() { _ = logFile.Close() })
	client := goplugin.NewClient(&goplugin.ClientConfig{
		HandshakeConfig:     handshake,
		VersionedPlugins:    map[int]goplugin.PluginSet{1: {"resource": &resourceplugin.GRPCPlugin{}}},
		GRPCBrokerMultiplex: true,
		GRPCDialOptions:     sdk.GRPCDialOptions(),
		Cmd:                 command,
		SkipHostEnv:         true,
		StartTimeout:        120 * time.Second,
		AllowedProtocols:    []goplugin.Protocol{goplugin.ProtocolGRPC},
		Logger:              hclog.New(&hclog.LoggerOptions{Name: p07PluginID, Output: logFile, Level: hclog.Debug}),
	})
	t.Cleanup(client.Kill)
	rpc, err := client.Client()
	require.NoError(t, err)
	backend := plugintypes.NewExternalBackend(client, rpc)
	reattach := client.ReattachConfig()
	rep.f("PLUGIN child started: binary=%s pid=%d protocol=%s addr=%s negotiated-version=%d", env.Binary, reattach.Pid, reattach.Protocol, reattach.Addr.String(), backend.NegotiatedVersion())
	return &p07Plugin{client: client, backend: backend}
}

type p07Harness struct {
	t        *testing.T

	rep      p07Report

func (h *p07Harness) fetch(ctx context.Context, reference resource.SchemaDocumentReference) (resource.SchemaDocument, error) {
	return h.ctrl.GetSchemaDocument(ctx, p07PluginID, h.route, reference)
}

func (h *p07Harness) descriptor(token string) (resource.ResourceType, bool) {
	for _, candidate := range h.catalog.Types {
		if candidate.Type == token {
			return candidate, true
		}
	}
	return resource.ResourceType{}, false
}

func (h *p07Harness) enumerate(token, representation string, limit int) ([]resource.Record, int, error) {
	target, err := h.ctrl.ResolveTarget(h.ctx, p07PluginID, ResolveTargetRequest{Route: h.route, Type: token, Representation: representation, Scope: resource.QueryScope{}})
	if err != nil {
		return nil, 0, fmt.Errorf("ResolveTarget: %s", p07Err(err))
	}
	request := HostEnumerateRequest{TargetID: target.TargetID, Request: resource.EnumerateRequest{Target: target.Target, PageSize: 100}}
	var records []resource.Record
	pages := 0
	for {
		page, err := h.ctrl.Enumerate(h.ctx, p07PluginID, request)
		if err != nil {
			return records, pages, fmt.Errorf("Enumerate page %d: %s", pages+1, p07Err(err))
		}
		pages++
		for _, record := range page.Records {
			records = append(records, record.Record)
		}
		if page.SourcePage.NextCursor == nil || len(records) >= limit {
			break
		}
		request.Request.Cursor = page.SourcePage.NextCursor
	}
	return records, pages, nil
}

type p07TypeResult struct {
	Label, Token, Representation string
	SchemaState, ResolveErr      string
	Validation                   string
	Unvalidated                  []resourcecatalog.Constraint
	Records, Admitted, Rejected  int
	Pages                        int
	EnumerateErr                 string
	AdmitValidation              map[string]int
	Messages                     map[string]int
	RejectSamples                map[string][]string
	MaxWorkRecord                string
	Work                         []int
	CoverageKinds                map[string]int
	UnavailableRecords           int
	UnavailableSamples           map[string]int
	RepresentationMismatch       int
}

func (r p07TypeResult) maxWork() int {
	maximum := 0
	for _, value := range r.Work {
		if value > maximum {
			maximum = value
		}
	}
	return maximum
}

func (r p07TypeResult) meanWork() float64 {
	if len(r.Work) == 0 {
		return 0
	}
	total := 0
	for _, value := range r.Work {
		total += value
	}
	return float64(total) / float64(len(r.Work))
}

// p07MinWork binary-searches the smallest successful Policy.MaxAdmissionWork,
// all other policy fields default, per contract section 9 "Admission work".
func p07MinWork(ctx context.Context, schema *resourcecatalog.Schema, body resource.Value, coverage resource.Coverage) (int, error) {
	policy := resourcecatalog.DefaultPolicy()
	low, high := 1, policy.MaxAdmissionWork
	policy.MaxAdmissionWork = high
	if _, err := resourcecatalog.Admit(ctx, schema, body, coverage, policy); err != nil {
		return 0, err
	}
	for low < high {
		middle := low + (high-low)/2
		policy.MaxAdmissionWork = middle
		if _, err := resourcecatalog.Admit(ctx, schema, body, coverage, policy); err == nil {
			high = middle
		} else if resource.ErrorCode(err) == "RESOURCE_LIMIT" {
			low = middle + 1
		} else {
			return 0, err
		}
	}
	return low, nil
}

func (h *p07Harness) admitRecords(label, token, representation string, records []resource.Record, schema *resourcecatalog.Schema, measureWork bool) p07TypeResult {
	result := p07TypeResult{
		Label: label, Token: token, Representation: representation, Records: len(records),
		AdmitValidation: map[string]int{}, Messages: map[string]int{}, RejectSamples: map[string][]string{}, CoverageKinds: map[string]int{}, UnavailableSamples: map[string]int{},
	}
	if schema != nil {
		result.Validation = schema.Validation
		result.Unvalidated = schema.Unvalidated
	} else {
		result.Validation = "none"
	}
	policy := resourcecatalog.DefaultPolicy()
	for index := range records {
		record := records[index]
		result.CoverageKinds[record.Coverage.Kind+fmt.Sprintf("%v", record.Coverage.Paths)]++
		if record.Representation.ID != representation {
			result.RepresentationMismatch++
		}
		admission, err := resourcecatalog.Admit(h.ctx, schema, record.Body, record.Coverage, policy)
		if err != nil {
			result.Rejected++
			message := resource.ErrorCode(err) + ": " + err.Error()
			result.Messages[message]++
			if len(result.RejectSamples[message]) < 3 {
				namespace := ""
				for _, entry := range record.Reference.Address.Scope {
					namespace += entry.Dimension + "=" + entry.Value + ","
				}
				result.RejectSamples[message] = append(result.RejectSamples[message], namespace+"id="+record.Reference.Address.ID)
			}
			continue
		}
		result.Admitted++
		result.AdmitValidation[admission.Validation]++
		if len(admission.UnavailablePaths) != 0 {
			result.UnavailableRecords++
			for _, path := range admission.UnavailablePaths {
				result.UnavailableSamples[path]++
			}
		}
		if measureWork {
			work, err := p07MinWork(h.ctx, schema, record.Body, record.Coverage)
			require.NoError(h.t, err, "%s %s", token, record.Reference.Address.ID)
			if work > result.maxWork() {
				scope := ""
				for _, entry := range record.Reference.Address.Scope {
					scope += entry.Dimension + "=" + entry.Value + ","
				}
				result.MaxWorkRecord = scope + "id=" + record.Reference.Address.ID
			}
			result.Work = append(result.Work, work)
		}
	}
	return result
}

var p07Index = regexp.MustCompile(`/[0-9]+`)

func (r p07TypeResult) print(rep p07Report, verbose bool) {
	status := r.SchemaState
	if r.ResolveErr != "" {
		status += " resolveErr=" + r.ResolveErr
	}
	rep.f("ROW %-14s rep=%-30s type=%-46s schema=%s validation=%s unvalidated=%d records=%d admitted=%d rejected=%d maxWork=%d meanWork=%.1f maxWorkRecord=%q pages=%d unavailableRecords=%d repMismatch=%d",
		r.Label, r.Representation, r.Token, status, r.Validation, len(r.Unvalidated), r.Records, r.Admitted, r.Rejected, r.maxWork(), r.meanWork(), r.MaxWorkRecord, r.Pages, r.UnavailableRecords, r.RepresentationMismatch)
	if r.EnumerateErr != "" {
		rep.f("     enumerateErr=%s", r.EnumerateErr)
	}
	coverage := make([]string, 0, len(r.CoverageKinds))
	for key, count := range r.CoverageKinds {
		coverage = append(coverage, fmt.Sprintf("%s x%d", key, count))
	}
	sort.Strings(coverage)
	if len(coverage) != 0 {
		rep.f("     recordCoverage: %s", strings.Join(coverage, "; "))
	}
	outcomes := make([]string, 0, len(r.AdmitValidation))
	for key, count := range r.AdmitValidation {
		outcomes = append(outcomes, fmt.Sprintf("%s x%d", key, count))
	}
	sort.Strings(outcomes)
	if len(outcomes) != 0 {
		rep.f("     admissionValidation: %s", strings.Join(outcomes, "; "))
	}
	if len(r.UnavailableSamples) != 0 {
		keys := make([]string, 0, len(r.UnavailableSamples))
		for key, count := range r.UnavailableSamples {
			keys = append(keys, fmt.Sprintf("%s x%d", key, count))
		}
		sort.Strings(keys)
		rep.f("     unavailablePaths: %s", strings.Join(keys, "; "))
	}
	messages := make([]string, 0, len(r.Messages))
	for message := range r.Messages {
		messages = append(messages, message)
	}
	sort.Strings(messages)
	for _, message := range messages {
		rep.f("     REJECT x%d %s", r.Messages[message], message)
		rep.f("       sample records: %v", r.RejectSamples[message])
	}
	if len(r.Messages) > 1 {
		normalized := map[string]int{}
		for message, count := range r.Messages {
			normalized[p07Index.ReplaceAllString(message, "/N")] += count
		}
		keys := make([]string, 0, len(normalized))
		for key := range normalized {
			keys = append(keys, key)
		}
		sort.Strings(keys)
		for _, key := range keys {
			rep.f("     REJECT-NORMALIZED x%d %s", normalized[key], key)
		}
	}
	if verbose {
		for _, constraint := range r.Unvalidated {
			rep.f("     unvalidated: pointer=%q keyword=%q", constraint.Pointer, constraint.Keyword)
		}
	}
}

func (h *p07Harness) resolveEntry(token, representation string) (*resourcecatalog.Entry, *resourcecatalog.Schema, error) {
	entry, err := h.snapshot.Entry(token, representation)
	if err != nil {
		return nil, nil, err
	}
	schema, err := entry.Resolve(h.ctx, h.fetch)
	return entry, schema, err
}

// runType enumerates every record of one catalog type and admits each body with
// the record's own Coverage against the schema resolved from the published binding.
func (h *p07Harness) runType(label, token, representation string, measureWork bool) (p07TypeResult, []resource.Record, *resourcecatalog.Schema) {
	entry, schema, resolveErr := h.resolveEntry(token, representation)
	records, pages, enumerateErr := h.enumerate(token, representation, 5000)
	var result p07TypeResult
	if resolveErr != nil {
		result = p07TypeResult{Label: label, Token: token, Representation: representation, Records: len(records), ResolveErr: p07Err(resolveErr), Messages: map[string]int{}, AdmitValidation: map[string]int{}, CoverageKinds: map[string]int{}, UnavailableSamples: map[string]int{}}
	} else {
		result = h.admitRecords(label, token, representation, records, schema, measureWork)
	}
	if entry != nil {
		result.SchemaState = entry.Binding.State
	}
	result.Pages = pages
	if enumerateErr != nil {
		result.EnumerateErr = enumerateErr.Error()
	}
	return result, records, schema
}

// controlSchema resolves the same binding against the served bytes with the
// Time/MicroTime nullable correction removed (harness-only mutation).
func (h *p07Harness) controlSchema(token, representation string) (*resourcecatalog.Schema, int, error) {
	entry, err := h.snapshot.Entry(token, representation)
	if err != nil {
		return nil, 0, err
	}
	if entry.Binding.DocumentRef == nil {
		return nil, 0, fmt.Errorf("binding has no document reference")
	}
	served, err := h.ctrl.GetSchemaDocument(h.ctx, p07PluginID, h.route, *entry.Binding.DocumentRef)
	if err != nil {
		return nil, 0, err
	}
	mutated, stripped, err := p07StripTimeNullable(served.Document)
	if err != nil {
		return nil, 0, err
	}
	catalog := resource.Catalog{Provider: h.catalog.Provider, Revision: "p07-control", Types: []resource.ResourceType{entry.Descriptor}}
	isolated := resourcecatalog.NewCatalog(resourcecatalog.DefaultPolicy())
	snapshot := isolated.Observe(resourcecatalog.ConnectionKey{PluginID: p07PluginID, Generation: "p07-control", Connection: "control", Epoch: 1}, catalog)
	controlEntry, err := snapshot.Entry(token, representation)
	if err != nil {
		return nil, 0, err
	}
	schema, err := controlEntry.Resolve(h.ctx, func(context.Context, resource.SchemaDocumentReference) (resource.SchemaDocument, error) {
		return resource.SchemaDocument{Reference: *entry.Binding.DocumentRef, Dialect: served.Dialect, Document: json.RawMessage(mutated)}, nil
	})
	return schema, stripped, err
}

func p07WithoutPath(value resource.Value, path ...string) resource.Value {
	out := resource.Value{Kind: "map", Map: make(map[string]resource.Value, len(value.Map))}
	for key, child := range value.Map {
		out.Map[key] = child
	}
	if len(path) == 1 {
		delete(out.Map, path[0])
		return out
	}
	if child, ok := out.Map[path[0]]; ok {
		out.Map[path[0]] = p07WithoutPath(child, path[1:]...)
	}
	return out
}

		publishedOutside, publishedOutsideOK := outside(published.Document)
		outsideEqual := nativeOutsideOK && publishedOutsideOK && bytes.Equal(nativeOutside, publishedOutside)
		expectedDelta := 0
		for _, pair := range [][2]string{{nativeTime, publishedTime}, {nativeMicro, publishedMicro}} {
			expectedDelta += len(pair[1]) - len(pair[0])
		}
		if outsideEqual {
			outsideEqualCount++
		}
		if reference.ID == "kubernetes.openapi.v3/v1" {
			rep.f("PHASE9 NATIVE   %s: %s", p07TimeComponent, nativeTime)
			rep.f("PHASE9 PUBLISHED %s: %s", p07TimeComponent, publishedTime)
			rep.f("PHASE9 NATIVE   %s: %s", p07MicroTimeComponent, nativeMicro)
			rep.f("PHASE9 PUBLISHED %s: %s", p07MicroTimeComponent, publishedMicro)
		}
		hasNullable := func(component string) bool {
			var fields map[string]json.RawMessage
			return json.Unmarshal([]byte(component), &fields) == nil && string(bytes.TrimSpace(fields["nullable"])) == "true"
		}
		timeCorrect := (!nativeTimePresent || hasNullable(publishedTime)) && (!nativeMicroPresent || hasNullable(publishedMicro))
		if same {
			identical++
		} else {
			failures++
		}
		if bytes.Equal(native, published.Document) {
			exact++
		}
		if nativeTimePresent || nativeMicroPresent {
			withTime++
			if timeCorrect {
				nullableOnBoth++
			}
		}
		rep.f("PHASE9 %-44s nativeBytes=%d publishedBytes=%d delta=%d componentBodyDelta=%d bytesOutsideTheTwoComponentsIdentical=%v semanticallyEqualAfterRemovingCorrection=%v exactBytesEqual=%v timeComponents(native/time=%v micro=%v) publishedNullableOnPresent=%v removedByStrip=%d revisionIsSha256OfPublished=%v",
			reference.ID, len(native), len(published.Document), len(published.Document)-len(native), expectedDelta, outsideEqual, same, bytes.Equal(native, published.Document), nativeTimePresent, nativeMicroPresent, timeCorrect, removed, reference.Revision == p07SHA(published.Document))
	}
	rep.f("PHASE9 SUMMARY documents=%d publishedTotalBytes=%d semanticallyEqualAfterRemovingCorrection=%d notEqual=%d exactBytesEqual=%d bytesOutsideTheTwoComponentsIdentical=%d documentsWithTimeComponents=%d publishedNullableOnAllPresent=%d", len(refList), totalServed, identical, failures, exact, outsideEqualCount, withTime, nullableOnBoth)

	// ---- Phase 10: sweep every other enumerable catalog type -------------------------------
	rep.f("PHASE10 === sweep: all other enumerable catalog types, default representation ===")
	sweepTotals := tally{}
	for _, descriptor := range catalog.Types {
		if seen[descriptor.Type] {
			continue
		}
		listable := false
		for _, capability := range descriptor.Capabilities {
			if capability.Operation == resource.OperationEnumerate {
				listable = true
			}
		}
		if !listable || descriptor.DefaultRepresentation == "" {
			continue
		}
		result, _, _ := h.runType(descriptor.DisplayName, descriptor.Type, descriptor.DefaultRepresentation, true)
		result.print(rep, false)
		sweepTotals.records += result.Records
		sweepTotals.admitted += result.Admitted
		sweepTotals.rejected += result.Rejected
		if result.maxWork() > sweepTotals.maxWork {
			sweepTotals.maxWork = result.maxWork()
		}
	}
	rep.f("PHASE10 SWEEP TOTALS records=%d admitted=%d rejected=%d maxWork=%d", sweepTotals.records, sweepTotals.admitted, sweepTotals.rejected, sweepTotals.maxWork)
	labels := make([]string, 0, len(table))
	for label := range table {
		labels = append(labels, label)
	}
	sort.Strings(labels)
	for _, label := range labels {
		rep.f("TABLE %-12s records=%d admitted=%d rejected=%d maxWork=%d", label, table[label].records, table[label].admitted, table[label].rejected, table[label].maxWork)
	}

	// ---- Phase 11: two connections to the same cluster -----------------------------------
	rep.f("PHASE11 === two more kubeconfig contexts (p07-a, p07-b) onto the SAME cluster ===")
	statsBefore := ctrl.schemas.Stats()
	routes := map[string]resource.AccessRoute{p07MainConn: route}
	for _, name := range []string{"p07-a", "p07-b"} {
		connectionStatus, err := ctrl.StartConnection(ctx, p07PluginID, name)
		require.NoError(t, err, name)
		rep.f("PHASE11 StartConnection(%s): status=%s", name, connectionStatus.Status)
		routes[name] = resource.AccessRoute{Connection: name}
	}
	perConnection := map[string]map[string]EditorSchemaAssociation{}
	perConnectionDocs := map[string]map[string]bool{}
	for name, access := range routes {
		list, err := ctrl.GetEditorSchemas(ctx, p07PluginID, access)
		require.NoError(t, err, name)
		perConnection[name] = map[string]EditorSchemaAssociation{}
		perConnectionDocs[name] = map[string]bool{}
		for _, association := range list {
			perConnection[name][association.Type+"|"+association.Key] = association
			if association.DocumentID != "" {
				perConnectionDocs[name][association.DocumentID] = true
			}
		}
		pod := perConnection[name]["k8s:/pods|"]
		rep.f("PHASE11 GetEditorSchemas(%s): associations=%d distinctDocumentIds=%d Pod: modelDirectory=%s fileMatch=%s documentId=%s status=%s", name, len(list), len(perConnectionDocs[name]), pod.ModelDirectory, pod.FileMatch, pod.DocumentID, pod.Status)
	}
	mainPod, aPod, bPod := perConnection[p07MainConn]["k8s:/pods|"], perConnection["p07-a"]["k8s:/pods|"], perConnection["p07-b"]["k8s:/pods|"]
	rep.f("PHASE11 Pod ModelDirectory distinct across connections=%v FileMatch distinct=%v DocumentID identical across connections=%v", mainPod.ModelDirectory != aPod.ModelDirectory && aPod.ModelDirectory != bPod.ModelDirectory && mainPod.ModelDirectory != bPod.ModelDirectory,
		mainPod.FileMatch != aPod.FileMatch && aPod.FileMatch != bPod.FileMatch && mainPod.FileMatch != bPod.FileMatch, mainPod.DocumentID == aPod.DocumentID && aPod.DocumentID == bPod.DocumentID)
	allDirectoriesDistinct := true
	directories := map[string]string{}
	for name, associations := range perConnection {
		for key, association := range associations {
			if association.ModelDirectory == "" {
				continue
			}
			if other, clash := directories[association.ModelDirectory]; clash {
				allDirectoriesDistinct = false
				rep.f("PHASE11 MODEL DIRECTORY CLASH %s: %s vs %s", association.ModelDirectory, other, name+"/"+key)
			}
			directories[association.ModelDirectory] = name + "/" + key
		}
	}
	rep.f("PHASE11 all (connection,type) ModelDirectories distinct=%v (count=%d)", allDirectoriesDistinct, len(directories))
	onlyInOne := 0
	for name, docs := range perConnectionDocs {
		for other, otherDocs := range perConnectionDocs {
			if name == other {
				continue
			}
			for digest := range docs {
				if !otherDocs[digest] {
					onlyInOne++
					rep.f("PHASE11 digest %s referenced by %s but not %s", digest, name, other)
				}
			}
		}
	}
	rep.f("PHASE11 digests referenced by one connection but not another: %d", onlyInOne)
	docMain, err := ctrl.GetEditorSchemaDocument(ctx, p07PluginID, routes[p07MainConn], mainPod.DocumentID)
	require.NoError(t, err)
	docB, err := ctrl.GetEditorSchemaDocument(ctx, p07PluginID, routes["p07-b"], mainPod.DocumentID)
	rep.f("PHASE11 GetEditorSchemaDocument(p07-b, digest first referenced via %s): err=%s bytesEqualToMain=%v", p07MainConn, p07Err(err), err == nil && bytes.Equal(docMain.Schema, docB.Schema))
	_, err = ctrl.GetEditorSchemaDocument(ctx, p07PluginID, routes["p07-a"], strings.Repeat("0", 64))
	rep.f("PHASE11 GetEditorSchemaDocument(p07-a, unreferenced digest): %s", p07Err(err))
	fieldsA, err := ctrl.GetSchemaFields(ctx, p07PluginID, SchemaFieldsRequest{Route: routes["p07-a"], Type: "k8s:/pods"})
	require.NoError(t, err)
	fieldsB, err := ctrl.GetSchemaFields(ctx, p07PluginID, SchemaFieldsRequest{Route: routes["p07-b"], Type: "k8s:/pods"})
	require.NoError(t, err)
	rep.f("PHASE11 GetSchemaFields Pod root: p07-a catalogRevision=%s schemaRevision=%s fields=%d | p07-b catalogRevision=%s schemaRevision=%s fields=%d | equal fields=%v", fieldsA.CatalogRevision, fieldsA.SchemaRevision, len(fieldsA.Fields), fieldsB.CatalogRevision, fieldsB.SchemaRevision, len(fieldsB.Fields), reflect.DeepEqual(fieldsA.Fields, fieldsB.Fields))
	statsAfter := ctrl.schemas.Stats()
	rep.f("PHASE11 controller catalog Stats: before(main only)=%+v after(3 connections)=%+v", statsBefore, statsAfter)
	_, err = ctrl.StopConnection(ctx, p07PluginID, "p07-b")
	require.NoError(t, err)
	rep.f("PHASE11 StopConnection(p07-b) ok; Stats=%+v", ctrl.schemas.Stats())
	_, err = ctrl.GetEditorSchemaDocument(ctx, p07PluginID, routes["p07-b"], mainPod.DocumentID)
	rep.f("PHASE11 GetEditorSchemaDocument(p07-b after StopConnection): %s", p07Err(err))
	docAfter, err := ctrl.GetEditorSchemaDocument(ctx, p07PluginID, routes["p07-a"], mainPod.DocumentID)
	rep.f("PHASE11 GetEditorSchemaDocument(p07-a still started, same digest): err=%s bytes=%d", p07Err(err), len(docAfter.Schema))

	rep.f("=== DONE elapsed=%s ===", time.Since(started).Round(time.Second))
}
