package main

import (
	"encoding/json"
	"os"
	"time"

	"github.com/omniviewdev/omniview/backend/pkg/invocation"
)

type debugCounter interface{ DebugCounts() map[string]int }

// inv3Counters writes the registries' counts to the file named by INV3_COUNTERS every 250 ms. INV3 proof probe (never committed).
func inv3Counters(realm *invocation.WindowRealm, ctrl any, others ...debugCounter) {
	path := os.Getenv("INV3_COUNTERS")
	if path == "" {
		return
	}
	for range time.Tick(250 * time.Millisecond) {
		out := realm.DebugCounts()
		if c, ok := ctrl.(interface{ DebugSourceCounts() map[string]int }); ok {
			for k, v := range c.DebugSourceCounts() {
				out[k] = v
			}
		}
		for _, other := range others {
			for k, v := range other.DebugCounts() {
				out[k] = v
			}
		}
		data, _ := json.Marshal(out)
		_ = os.WriteFile(path+".tmp", data, 0o600)
		_ = os.Rename(path+".tmp", path)
	}
}

package invocation

// DebugCounts is an INV3 proof probe (never committed).
func (r *WindowRealm) DebugCounts() map[string]int {
	r.mu.Lock()
	defer r.mu.Unlock()
	return map[string]int{"realmScopes": len(r.scopes), "realmClosed": len(r.closed), "realmRefused": len(r.refused), "realmSubscribers": len(r.subscribers)}
}

	s.mu.Lock()
	defer s.mu.Unlock()
	return map[string]int{"graphWindows": len(s.windows), "graphObservations": len(s.observations)}

// DebugSourceCounts is an INV3 proof probe (never committed).
func (c *controller) DebugSourceCounts() map[string]int {
	sc := c.sources
	sc.mu.Lock()
	defer sc.mu.Unlock()
	return map[string]int{"sourceSessions": len(sc.sessions), "sourceViews": len(sc.views), "sourceInterests": len(sc.interests), "retiredOwners": len(sc.retiredOwners)}
}
