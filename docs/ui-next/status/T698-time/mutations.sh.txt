#!/bin/bash
# Negative controls for T698-time (#698 item 23). Each fault is put into a SCRATCH copy of one source file and built with
# `go test -overlay`; the worktree is never edited. A control has two halves (AGENTS 28):
#   1. with the touched tests, the ones that go red are exactly the ones named for it;
#   2. with the whole package and the named tests skipped (the assertion removed), the same fault is reported: PASS
#      (only the named tests see it) or ALSO CAUGHT by the other tests of the package that see it (not a failure of the
#      control: those tests were there before and were not touched).
# TestAShellThatIsNotInstalledFallsBack... fails on a clean origin/main of this machine (no shell fallback here): it is
# skipped in half 2 so that it does not read as a detection.
# Run from anywhere in the worktree:  bash docs/ui-next/status/T698-time/mutations.sh.txt [control-name]
# Needs the Go toolchain on PATH.
export GOWORK=off GOTOOLCHAIN=local
WT=$(git rev-parse --show-toplevel) || exit 2
PKG=./backend/pkg/terminal/
M=$(mktemp -d "${TMPDIR:-/tmp}/t698time-mut.XXXXXX") || { echo "no scratch directory" >&2; exit 2; }
trap 'rm -rf -- "$M"' EXIT
cd "$WT" || exit 2
only="${1:-}"
bad=0
TOUCHED='^(TestClosingASessionThatIsPausedForCreditEndsItAndDeliversItsClose|TestShutdownEndsEverySessionAndRefusesNewOnes|TestClosingHangsUpAtOnceAndKillsWhatIgnoresTheHangupOnlyAfterTheGrace|TestAWindowGoingAwayDetachesATerminalWithAPageKeyAndEndsAnyOther|TestDetachKeepsTheTerminalForALeaseThatIsCutToTheMostAndIsIdempotent|TestALeaseThatRunsOutEndsTheTerminalTheWayACloseDoes)$'
BASELINE=TestAShellThatIsNotInstalledFallsBackAndTheAnswerSaysWhichStarted

# name | file | sed expression | the tests named for it (sorted, separated by spaces)
run() {
  name="$1"; file="$2"; expr="$3"; named="$4"
  [ -n "$only" ] && [ "$only" != "$name" ] && return
  cp "$WT/$file" "$M/$name.go"
  sed -i -E -e "$expr" "$M/$name.go"
  if cmp -s "$WT/$file" "$M/$name.go"; then echo "CONTROL $name: FAULT NOT APPLIED"; bad=1; return; fi
  printf '{"Replace":{"%s":"%s"}}' "$WT/$file" "$M/$name.go" > "$M/$name.json"

  out=$(go test -overlay="$M/$name.json" "$PKG" -run "$TOUCHED" -count=1 -timeout 300s 2>&1)
  code=$?
  failed=$(echo "$out" | grep -E '^--- FAIL' | sed -E 's/--- FAIL: ([A-Za-z0-9_]+).*/\1/' | sort | tr '\n' ' ' | sed 's/ $//')
  if [ $code -eq 0 ]; then half1="UNDETECTED (every test passed)"; bad=1
  elif [ -z "$failed" ]; then half1="NOT A DETECTION (the build or the setup failed)"; bad=1
  elif [ "$failed" != "$named" ]; then half1="RED in [$failed], named [$named]"; bad=1
  else half1="RED in exactly [$failed]"; fi

  skip="^($(echo "$named $BASELINE" | sed 's/ /|/g'))\$"
  out2=$(go test -overlay="$M/$name.json" "$PKG" -skip "$skip" -count=1 -timeout 600s 2>&1)
  if [ $? -eq 0 ]; then half2="without them: PASS (undetected)"
  else half2="without them: ALSO CAUGHT by other tests [$(echo "$out2" | grep -E '^--- FAIL' | sed -E 's/ \(.*//' | tr '\n' ' ')]"; fi
  echo "CONTROL $name: $half1 | $half2"
}

R=backend/pkg/terminal/reattach.go
G=backend/pkg/terminal/manager.go
LEASE='TestAWindowGoingAwayDetachesATerminalWithAPageKeyAndEndsAnyOther TestDetachKeepsTheTerminalForALeaseThatIsCutToTheMostAndIsIdempotent'
RUNOUT=TestALeaseThatRunsOutEndsTheTerminalTheWayACloseDoes
HANG=TestClosingHangsUpAtOnceAndKillsWhatIgnoresTheHangupOnlyAfterTheGrace
PAUSED=TestClosingASessionThatIsPausedForCreditEndsItAndDeliversItsClose
SHUT=TestShutdownEndsEverySessionAndRefusesNewOnes

run lease-end-read-from-the-wall-clock $R \
  's/^\ts.leaseEnd = s.clock.Now\(\).Add\(lease\)$/\ts.leaseEnd = time.Now().Add(lease)/' "$LEASE"
run lease-end-one-second-late $R \
  's/^\ts.leaseEnd = s.clock.Now\(\).Add\(lease\)$/\ts.leaseEnd = s.clock.Now().Add(lease + time.Second)/' "$LEASE"
run lease-timer-fires-early $R \
  's/s.clock.AfterFunc\(lease, func\(\) \{ s.expire\(gen\) \}\)/s.clock.AfterFunc(lease\/2, func() { s.expire(gen) })/' "$RUNOUT"
run lease-timer-fires-late $R \
  's/s.clock.AfterFunc\(lease, func\(\) \{ s.expire\(gen\) \}\)/s.clock.AfterFunc(lease*2, func() { s.expire(gen) })/' "$RUNOUT"
run close-does-not-wait-for-the-session $G \
  's/^\tm.awaitGone\(\[\]\*session\{s\}\)$/\t_ = s/' "$HANG"
run shutdown-does-not-wait-for-the-sessions $G \
  's/^\tm.awaitGone\(sessions\)$/\t_ = sessions/' "$SHUT"
run close-does-not-release-the-pacing $G \
  's/^\ts.flow.release\(\)$/\t_ = 0/' "$PAUSED"
run hangup-does-not-reach-the-shell $G \
  's/^\t_ = s.cmd.Process.Signal\(syscall.SIGHUP\)$/\t_ = 0/' "$HANG"
run grace-never-kills $G \
  '/^func \(s \*session\) hangup/,/^}/{s/^\ts.signalForeground\(syscall.SIGKILL\)$/\t_ = 0/;s/^\ts.cancel\(\)$/\t_ = 0/}' "$RUNOUT $HANG"
run grace-is-no-grace $G \
  's/^\ttimer := time.NewTimer\(grace\)$/\ttimer := time.NewTimer(0)/' "$HANG"

exit $bad
