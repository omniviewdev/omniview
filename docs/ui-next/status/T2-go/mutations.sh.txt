#!/bin/bash
# Negative controls for T2-go: each fault is put into a SCRATCH copy of one source file and built with
# `go test -overlay`; the worktree is never edited. A control passes when the test named for it turns red.
# ROOT is the worktree (default: the current directory); ONLY=<regexp> runs the controls whose name matches.
export GOWORK=off GOTOOLCHAIN=local
WT=${ROOT:-$PWD}
M=${MDIR:-/tmp/t2go-mut}
ONLY=${ONLY:-.}
rm -rf "$M" && mkdir -p "$M"
cd "$WT" || exit 2

# name | package | file | sed expression | test regexp
run() {
  name="$1"; pkg="$2"; file="$3"; expr="$4"; test="$5"
  echo "$name" | grep -Eq "$ONLY" || return
  cp "$WT/$file" "$M/$name.go"
  sed -i '' -e "$expr" "$M/$name.go"
  if cmp -s "$WT/$file" "$M/$name.go"; then echo "CONTROL $name: FAULT NOT APPLIED"; return; fi
  printf '{"Replace":{"%s":"%s"}}' "$WT/$file" "$M/$name.go" > "$M/$name.json"
  out=$(go test -overlay="$M/$name.json" "$pkg" -run "$test" -count=1 -timeout 150s 2>&1)
  code=$?
  failed=$(echo "$out" | grep -E "^--- FAIL" | sed 's/ (.*//' | tr '\n' ' ')
  if [ $code -eq 0 ]; then echo "CONTROL $name: UNDETECTED (tests passed) <- $test"
  elif [ -z "$failed" ]; then echo "CONTROL $name: NOT A DETECTION (no test failed: the build or the setup did)"
  else echo "CONTROL $name: RED  $failed"; fi
}
