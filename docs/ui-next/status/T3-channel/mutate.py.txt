#   python3 docs/ui-next/status/T3-channel/mutate.py.txt [control number ...]
#
# Every control is a fault in a SCRATCH copy of one non-test source file (go test -overlay): the worktree is not edited. For each
# control the script runs the tests named for it twice:
#   1. with the fault: every named test must fail, by an assertion of its own (a testify "Error Trace" and the message written
#      here), no other test of the run may fail, and nothing may panic, time out or fail to build;
#   2. with the fault and the assertions that catch it switched off (guarded() in the tests reads TERMINAL_CHANNEL_CONTROL_OFF):
#      every named test must pass, which shows that nothing else of that test catches the fault.
# The faults are the ones a careless change would make; each is the smallest edit that breaks the property. Other tests of the
# package may catch the same fault as well: a control says that the named test does, by that assertion, and that it is the only
# thing in that test that does.

def read(path):
    with open(os.path.join(ROOT, path)) as handle:
        return handle.read()


def replaced(text, old, new):
    if text.count(old) != 1:
        sys.exit("control cannot be built: %r is in the file %d times" % (old[:80], text.count(old)))
    return text.replace(old, new)


def run(package, tests, overlay_files, off):
    scratch = tempfile.mkdtemp(prefix="t3-control-")
    replace = {}
    for path, text in overlay_files.items():
        copy = os.path.join(scratch, path.replace("/", "__"))
        with open(copy, "w") as handle:
            handle.write(text)
        replace[os.path.join(ROOT, path)] = copy
    overlay = os.path.join(scratch, "overlay.json")
    with open(overlay, "w") as handle:
        json.dump({"Replace": replace}, handle)
    env = dict(os.environ, GOWORK="off", GOTOOLCHAIN="local")
    env.pop(OFF, None)
    if off:
        env[OFF] = ",".join(off)
    done = subprocess.run(
        ["go", "test", "-overlay=" + overlay, "-run", "^(" + "|".join(tests) + ")$", "-count=1", "-timeout", "300s", "-v",
         "./" + package + "/"],
        cwd=ROOT, env=env, capture_output=True, text=True, timeout=900)
    return done.returncode, done.stdout + done.stderr


def block(out, name):
    start = out.find("=== RUN   " + name)
    end = out.find("--- FAIL: " + name)
    return out[start:end] if 0 <= start < end else out


def red_as_named(out, named):
    failed = set(re.findall(r"^--- FAIL: (\w+)", out, re.M))
    if failed != {test for test, _ in named}:
        return False, "failed %s, named %s" % (sorted(failed), sorted(test for test, _ in named))
    for marker in ("panic:", "test timed out", "[build failed]", "[setup failed]"):
        if marker in out:
            return False, "the run has %r: not an assertion" % marker
    for test, message in named:
        text = block(out, test)
        if "Error Trace:" not in text:
            return False, "%s failed without a testify assertion" % test
        if message is not None and message not in text:
            return False, "%s failed, but not with %r" % (test, message)
    return True, ""


def main():
    chosen = {int(arg) for arg in sys.argv[1:]}
    bad = 0
    for number, what, path, edits, checks in CONTROLS:
        if chosen and number not in chosen:
            continue
        faulty = read(path)
        for old, new in edits:
            faulty = replaced(faulty, old, new)
        print("C%-2d %s" % (number, what), flush=True)
        for package, named, guards in checks:
            tests = [test for test, _ in named]
            code, out = run(package, tests, {path: faulty}, [])
            ok, why = red_as_named(out, named)
            print("      %-8s with the fault: %s" % (package.split("/")[-1], "RED by its assertion: %s" % ", ".join(tests) if ok else "NOT AS NAMED (%s)" % why), flush=True)
            if not ok:
                bad += 1
                print(out[-2500:])
            code, out = run(package, tests, {path: faulty}, guards)
            passed = code == 0 and all(("--- PASS: " + test) in out for test in tests)
            print("      %-8s fault, %s off: %s" % ("", ", ".join(guards), "passes (nothing else of the test catches it)" if passed else "STILL FAILS (exit %d)" % code), flush=True)
            if not passed:
                bad += 1
                print(out[-2500:])
    print("controls not as named:", bad)
    sys.exit(1 if bad else 0)


main()
