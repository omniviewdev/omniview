let caught = 0;
for (const mutant of mutants) {
  const dir = mkdtempSync(path.join(tmpdir(), 'kl3-mutant-'));
  const work = path.join(dir, 'scripts', 'ui-next-pods');
  cpSync(source, work, { recursive: true });
  symlinkSync(nodeModules, path.join(dir, 'node_modules'));
  const target = path.join(work, mutant.file);
  const text = readFileSync(target, 'utf8');
  if (!text.includes(mutant.find)) {
    console.log(`SKIP ${mutant.id}: the text to change is not in ${mutant.file}`);
    rmSync(dir, { recursive: true, force: true });
    continue;
  }
  writeFileSync(target, text.replace(mutant.find, mutant.replace));
  const run = spawnSync('node', ['--test', '--test-reporter=tap', path.join(work, mutant.test)], { encoding: 'utf8' });
  const failed = [...run.stdout.matchAll(/^\s*not ok \d+ - (.*)$/gm)].map((match) => match[1]);
  const named = failed.some((name) => name.includes(mutant.named));
  if (failed.length > 0 && named) caught += 1;
  console.log(`${failed.length > 0 && named ? 'CAUGHT' : failed.length > 0 ? 'RED-OTHER' : 'SURVIVED'} ${mutant.id} -> ${failed.length} red: ${failed.map((name) => name.slice(0, 70)).join(' | ')}`);
  rmSync(dir, { recursive: true, force: true });
}
console.log(`${caught} of ${mutants.length} caught by the control named for the rule`);
