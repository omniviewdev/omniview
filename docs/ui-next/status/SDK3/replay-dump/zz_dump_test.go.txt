package logs

// THROWAWAY (SDK3): dumps, for one session of a fixture source, the log frames the plugin put on the wire and the payloads the host's lane emitted
// to the window, normalised (session ID and event times set aside, events in a fixed order). Run on the host at SDK v0.21.0 (plain handler) and at v0.22.0
// (a handler that declares ReplayCount, which this host never asks for) and diff the two files.

type dumpFrames struct {
	mu     sync.Mutex
	frames []*logspb.LogStreamOutput
}

type dumpStream struct {
	grpc.ServerStream
	rec *dumpFrames
}

func (s *dumpStream) SendMsg(m any) error {
	if f, ok := m.(*logspb.LogStreamOutput); ok {
		s.rec.mu.Lock()
		s.rec.frames = append(s.rec.frames, proto.Clone(f).(*logspb.LogStreamOutput))
		s.rec.mu.Unlock()
	}
	return s.ServerStream.SendMsg(m)
}

func TestDumpTheStreamOfAFixtureSource(t *testing.T) {
	out := os.Getenv("SDK3_DUMP")
	if out == "" {
		t.Skip("SDK3_DUMP names the file this throwaway test writes")
	}
	ctx, cancel := context.WithTimeout(t.Context(), 30*time.Second)
	t.Cleanup(cancel)
	lines := []string{"2026-11-22T10:00:00Z first line", "2026-11-22T10:00:01Z second line", "2026-11-22T10:00:02Z third line"}
	h := sdklogs.Handler{
		Plugin: "test", Resource: "pod",
		Handler: func(*sdktypes.PluginContext, sdklogs.LogStreamRequest) (io.ReadCloser, error) {
			return io.NopCloser(strings.NewReader(strings.Join(lines, "\n") + "\n")), nil
		},
		SourceBuilder: logtest.SingleSourceBuilder("pod", map[string]string{"source": "pod"}),
	}
	declareReplayCount(&h)
	manager := sdklogs.NewManager(sdklogs.ManagerConfig{Logger: logging.NewNop(), Clock: timeutil.NewInstantFakeClock(), Handlers: map[string]sdklogs.Handler{"test/pod": h}})
	manager.BindConnectionContextProvider(aLeaseThatRunsTheCallback{})
	t.Cleanup(manager.Close)
	rec := &dumpFrames{}
	server := grpc.NewServer(sdkutils.RegisterServerOpts([]grpc.ServerOption{grpc.ChainStreamInterceptor(func(srv any, ss grpc.ServerStream, info *grpc.StreamServerInfo, handler grpc.StreamHandler) error {
		if strings.HasSuffix(info.FullMethod, "/Stream") {
			ss = &dumpStream{ServerStream: ss, rec: rec}
		}
		return handler(srv, ss)
	})}, logging.NewNop())...)
	plugin := &sdklogs.Plugin{Impl: manager}
	require.NoError(t, plugin.GRPCServer(nil, server))
	listener := bufconn.Listen(1024 * 1024)
	go func() { _ = server.Serve(listener) }()
	t.Cleanup(func() { server.Stop(); _ = listener.Close() })
	transport, err := grpc.DialContext(ctx, "passthrough:///dump",
		grpc.WithContextDialer(func(ctx context.Context, _ string) (net.Conn, error) { return listener.DialContext(ctx) }),
		grpc.WithTransportCredentials(insecure.NewCredentials()),
		grpc.WithChainUnaryInterceptor(sdk.ClientPluginContextInterceptor(logging.NewNop())),
	)
	require.NoError(t, err)
	t.Cleanup(func() { _ = transport.Close() })
	client, err := plugin.GRPCClient(ctx, nil, transport)
	require.NoError(t, err)
	backend := plugintypes.NewInProcessBackend(plugintypes.InProcessBackendConfig{Providers: map[string]interface{}{"log": client}})
	window := newEventRecorder()
	ctrl := NewController(logging.NewNop(), nil, immediateUIConnection{}).(*controller)
	ctrl.ctx = ctx
	ctrl.emitter = window
	t.Cleanup(func() { cancel(); _ = ctrl.OnPluginStop("p", config.PluginMeta{ID: "p"}) })
	require.NoError(t, ctrl.OnPluginStart("p", config.PluginMeta{ID: "p"}, backend))
	go ctrl.runMux()
	session, err := ctrl.CreateSession(ctx, "p", "route", sdklogs.CreateSessionOptions{ResourceKey: "pod", ResourceID: "pod", Options: sdklogs.LogSessionOptions{IncludeTimestamps: true, IncludeSourceEvents: true}})
	require.NoError(t, err)
	window.linesOf(t, session.ID, len(lines))
	require.Eventually(t, func() bool {
		for _, e := range window.recorded() {
			if e.key == "core/logs/event/"+session.ID && strings.Contains(e.data[0].(string), `"type":5`) {
				return true
			}
		}
		return false
	}, 10*time.Second, 5*time.Millisecond)

	var dump strings.Builder
	marshal := proto.MarshalOptions{Deterministic: true}
	var wireLines, wireEvents []string
	rec.mu.Lock()
	for _, f := range rec.frames {
		if f.GetSessionId() != session.ID {
			continue
		}
		n := proto.Clone(f).(*logspb.LogStreamOutput)
		n.SessionId = ""
		if l := n.GetLine(); l != nil {
			l.SessionId = ""
			b, _ := marshal.Marshal(n)
			wireLines = append(wireLines, hex.EncodeToString(b))
		} else if e := n.GetEvent(); e != nil {
			e.Timestamp = timestamppb.New(time.Time{})
			b, _ := marshal.Marshal(n)
			wireEvents = append(wireEvents, hex.EncodeToString(b))
		}
	}
	rec.mu.Unlock()
	slices.Sort(wireEvents)
	for _, l := range wireLines {
		fmt.Fprintf(&dump, "wire line  %s\n", l)
	}
	for _, e := range wireEvents {
		fmt.Fprintf(&dump, "wire event %s\n", e)
	}
	var windowLines, windowEvents []string
	for _, e := range window.recorded() {
		payload := e.data[0].(string)
		switch e.key {
		case "core/logs/lines/" + session.ID:
			var batch []sdklogs.LogLine
			require.NoError(t, json.Unmarshal([]byte(payload), &batch))
			for _, l := range batch {
				l.SessionID = ""
				b, _ := json.Marshal(l)
				windowLines = append(windowLines, string(b))
			}
			if bytes.Contains([]byte(payload), []byte(`"replay"`)) {
				windowLines = append(windowLines, "PAYLOAD HAS replay KEY")
			}
		case "core/logs/event/" + session.ID:
			var ev sdklogs.LogStreamEvent
			require.NoError(t, json.Unmarshal([]byte(payload), &ev))
			ev.Timestamp = time.Time{}
			b, _ := json.Marshal(ev)
			windowEvents = append(windowEvents, string(b))
		}
	}
	slices.Sort(windowEvents)
	for _, l := range windowLines {
		fmt.Fprintf(&dump, "window line  %s\n", l)
	}
	for _, e := range windowEvents {
		fmt.Fprintf(&dump, "window event %s\n", e)
	}
	require.NoError(t, os.WriteFile(out, []byte(dump.String()), 0o644))
}
