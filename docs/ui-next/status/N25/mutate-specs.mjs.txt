// The floor: every control of every unit file taken out, and every unit.
const floor = [];
const dir = 'scripts/ci/probe-units';
for (const name of readdirSync(path.join(origin, dir)).filter((n) => n.endsWith('.mjs')).sort()) {
  const file = `${dir}/${name}`;
  const text = readFileSync(path.join(origin, file), 'utf8');
  const count = (s) => text.split(s).length - 1;
  // every `controls: [...]` list (a list may wrap over lines): its quoted names
  for (const match of text.matchAll(/controls:\s*\[([^\]]*)\]/g)) {
    for (const nameMatch of match[1].matchAll(/'([^']+)'/g)) {
      const control = nameMatch[1];
      const list = match[0];
      const without = list.replace(new RegExp(`'${control.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}',?\\s*`), '').replace(/,\s*\]$/, ']');
      if (without === list || count(list) !== 1) continue;
      floor.push({ id: `F-control-${name.replace('.mjs', '')}-${control}`, area: 'floor', file, find: list, replace: without });
    }
  }
  // every unit object
  for (const match of text.matchAll(/\n  \{\n    id: '([^']+)',[\s\S]*?\n  \},?(?=\n)/g)) {
    floor.push({ id: `F-unit-${match[1]}`, area: 'floor', file, find: match[0], replace: '' });
  }
}
export default [...engine, ...workflow, ...floor];
