// Dev tool (not committed): which repository source files render into each fixture entry's bundle (vite build, rollup bundle modules with renderedLength > 0).
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { build, loadConfigFromFile, mergeConfig } from 'vite';

const repo = process.argv[2];
const which = process.argv[3]; // shell | page
const mode = process.argv[4] ?? 'production';
const out = process.argv[5];
const here = path.join(repo, which === 'shell' ? 'ui-next/shell/probes' : 'ui-next/page/probes');
const inputs = which === 'shell'
  ? ['fixture.html', 'host-fixture.html', 'palette-fixture.html', 'palette-plugins-fixture.html', 'notices-fixture.html', 'logs-open-fixture.html'].map((n) => path.join(here, n))
  : [path.join(here, 'fixture.html')];
const loaded = await loadConfigFromFile({ command: 'build', mode }, path.join(repo, 'vite.config.ts'));
const result = {};
const capture = {
  name: 'capture-inputs',
  generateBundle(_o, bundle) {
    const chunks = Object.values(bundle).filter((c) => c.type === 'chunk');
    const byName = new Map(chunks.map((c) => [c.fileName, c]));
    for (const c of chunks) {
      if (!c.isEntry) continue;
      const seen = new Set();
      const files = new Set();
      const walk = (chunk) => {
        if (seen.has(chunk.fileName)) return;
        seen.add(chunk.fileName);
        for (const [id, m] of Object.entries(chunk.modules)) {
          if (m.renderedLength > 0 && !id.includes('node_modules') && !id.startsWith('\0')) files.add(path.relative(repo, id.split('?')[0]));
        }
        for (const imp of [...chunk.imports, ...(chunk.dynamicImports ?? [])]) { const n = byName.get(imp); if (n) walk(n); }
      };
      walk(c);
      result[path.relative(repo, c.facadeModuleId ?? c.fileName)] = [...files].sort();
    }
  },
};
await build(mergeConfig(loaded.config, { configFile: false, logLevel: 'error', mode, plugins: [capture], build: { outDir: path.join(out, which + '-' + mode), emptyOutDir: true, ...(mode === 'development' ? { minify: false } : {}), rollupOptions: { input: inputs } } }));
import { writeFileSync } from 'node:fs';
writeFileSync(path.join(out, `inputs-${which}-${mode}.json`), JSON.stringify(result, null, 1));
for (const [k, v] of Object.entries(result)) console.log(k, v.length);
