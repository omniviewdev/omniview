def run(files):
    r = subprocess.run(['node', '--test'] + files, cwd=ROOT, capture_output=True, text=True)
    out = r.stdout + r.stderr
    failing = [l.strip()[2:].strip() for l in out.splitlines() if l.strip().startswith('✖') and ' (' in l and not l.strip().startswith('✖ failing')]
    failing = list(dict.fromkeys([re.sub(r' \([0-9.]+ms\)$', '', f) for f in failing]))
    return r.returncode, failing
only = sys.argv[1] if len(sys.argv) > 1 else None
# baseline
alltests = sorted(set(sum([m[4] for m in M], [])))
code, failing = run(alltests)
print('baseline exit', code, 'failing', failing)
if code != 0: sys.exit(1)
tripped = 0; results = []
for name, f, old, new, tests in M:
    if only and only not in name: continue
    path = os.path.join(ROOT, f)
    src = open(path).read()
    if src.count(old) != 1:
        print('SKIP (pattern count %d): %s' % (src.count(old), name)); results.append((name, 'pattern-miss', [])); continue
    open(path, 'w').write(src.replace(old, new))
    try:
        code, failing = run(tests)
    finally:
        open(path, 'w').write(src)
    ok = code != 0
    tripped += ok
    results.append((name, 'tripped' if ok else 'SURVIVED', failing[:3]))
    print(('TRIPPED ' if ok else 'SURVIVED ') + name + ' -> ' + ('; '.join(x[:70] for x in failing[:2])))
print('mutations', len(results), 'tripped', tripped, 'survived', sum(1 for r in results if r[1] == 'SURVIVED'), 'pattern-miss', sum(1 for r in results if r[1] == 'pattern-miss'))
json.dump(results, open('<a clone of the committed tree with node_modules linked>ation-results.json', 'w'), indent=1)
