// Dev tool (not committed): the in-page calibration against the centre's open, per engine, several trials, with the machine's load.
import os from 'node:os';
import { ENGINES, launch } from '<worktree>/packages/kit/probes/probe-lib.mjs';
import { NATIVE_VIEWPORTS, buildFixtures, open, serveFixtures, sleep } from '<worktree>/ui-next/shell/probes/harness.mjs';

const BELL = '[data-notices-part="bell"]';
const CENTRE = '[data-notices-part="centre"]';
const TRIALS = Number(process.env.TRIALS ?? 6);
const median = (xs) => { const s = [...xs].sort((a, b) => a - b); const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };

function calibrate() {
  const host = document.createElement('div');
  host.style.cssText = 'position:fixed;left:0;top:0;width:560px;height:480px;overflow:hidden;visibility:hidden;contain:strict';
  document.body.append(host);
  const once = () => {
    const fragment = document.createDocumentFragment();
    for (let row = 0; row < 50; row += 1) {
      const line = document.createElement('div');
      line.style.cssText = 'display:flex;gap:8px;padding:4px 8px';
      for (let cell = 0; cell < 8; cell += 1) {
        const span = document.createElement('span');
        span.textContent = `row ${row} cell ${cell} lorem ipsum`;
        span.style.cssText = cell === 1 ? 'flex:1 1 auto' : 'flex:0 0 auto';
        line.append(span);
      }
      fragment.append(line);
    }
    host.replaceChildren(fragment);
    host.getBoundingClientRect();
    host.lastElementChild.getBoundingClientRect();
  };
  const REPEAT = 8;
  const sample = () => {
    const start = performance.now();
    for (let i = 0; i < REPEAT; i += 1) once();
    return (performance.now() - start) / REPEAT;
  };
  for (let i = 0; i < 3; i += 1) sample();
  const samples = Array.from({ length: 9 }, sample).sort((a, b) => a - b);
  host.remove();
  return samples[4];
}

function openClock() {
  const opens = (window.__opens = []);
  let from = null;
  window.addEventListener('click', (event) => { if (event.target instanceof Element && event.target.closest('[data-notices-part="bell"]') !== null) from = performance.now(); }, true);
  window.addEventListener('click', () => {
    if (from === null) return;
    const box = document.querySelector('[data-notices-part="centre"]')?.getBoundingClientRect();
    const ms = performance.now() - from;
    from = null;
    opens.push({ ms, found: box !== undefined && box.height > 0 });
  });
}

const production = await serveFixtures(buildFixtures('production'));
for (const engineName of (process.env.ENGINES ?? 'chromium,webkit').split(',')) {
  const browser = await launch(engineName);
  for (let trial = 0; trial < TRIALS; trial += 1) {
    const { page, context } = await open(browser, `${production.origin}/shell/probes/notices-fixture.html?case=real&centre=1&seed=many`, NATIVE_VIEWPORTS['1440x900'], { init: `(${openClock.toString()})()` });
    await sleep(600);
    const load = os.loadavg()[0];
    const cals = [];
    for (let i = 0; i < 5; i += 1) cals.push(await page.evaluate(calibrate));
    for (let i = 0; i < 9; i += 1) {
      await page.click(BELL);
      await page.waitForSelector(CENTRE, { timeout: 5000 });
      await sleep(150);
      await page.keyboard.press('Escape');
      await page.waitForSelector(CENTRE, { state: 'detached', timeout: 5000 });
      await sleep(150);
    }
    const opens = (await page.evaluate(() => window.__opens)).map((o) => o.ms);
    const openMedian = median(opens);
    const cal = median(cals);
    console.log(`${engineName} trial ${trial}: load ${load.toFixed(1)}  calibration ${cal.toFixed(2)} ms (min ${Math.min(...cals).toFixed(2)})  open median ${openMedian.toFixed(1)} ms  ratio open/cal ${(openMedian / cal).toFixed(2)}`);
    await context.close();
  }
  await browser.close();
}
production.server.close();
