// T1 renderer bench: the DRIVER. Throwaway harness (suffix .txt so no tool compiles or lints it).
//
// Which xterm.js renderer (built-in DOM, or the WebGL addon) should the host's Terminal use, and what does the terminal cost?
// Questions B1 to B10 of the T1 renderer-bench task; the page side is renderer-bench-page.txt.
//
// RUN (from the worktree root, ui-next-t1-terminal; Playwright 1.64.1 browsers webkit + chromium installed):
//   mkdir -p /tmp/t1-bench && cp docs/ui-next/status/T1/renderer-bench.mjs.txt /tmp/t1-bench/renderer-bench.mjs
//   node /tmp/t1-bench/renderer-bench.mjs                       # all questions, all three engines (cwd = the worktree root; the page source is read from docs/ui-next/status/T1/renderer-bench-page.txt)
//   node /tmp/t1-bench/renderer-bench.mjs --only B3,B8 --engines webkit,chromium-gpu
//   node /tmp/t1-bench/renderer-bench.mjs --list                # the question ids
// Options (command line only, no environment variables):
//   --engines a,b   webkit | chromium | chromium-gpu   (default: all three)
//                   chromium     = probe-lib launch: Playwright's headless shell, args --font-render-hinting=none. Its WebGL is SwiftShader (software).
//                   chromium-gpu = channel 'chromium' (new headless, the full browser), the same arg; WebGL is ANGLE/Metal on the real GPU (what WebView2 would have).
//   --only B1,B3    questions to run (default all)
//   --out file      JSON result file; results are MERGED into it, so questions can be run one by one
//                   (default <tmp>/t1-bench/assets/renderer-bench.json, <tmp> being the real path of /tmp)
//   --assets dir    screenshots (default <tmp>/t1-bench/assets/renderer-bench-shots)
//   --reps n        repetitions where a question has them (default: per question)
//   --w2 0|1        run the 82 MB flood W2 (default 1)
// The driver builds the page with the root's vite (production build, esbuild minify) into /tmp/t1-bench/dist (outside the repo), serves it
// from a local static server (cross-origin-isolated headers, for performance.measureUserAgentSpecificMemory), and reads the machine's
// 1-minute load average before each run; above 30 it waits (the machine is shared). Needs macOS (ps).

// ------------------------------------------------------------------ load average
const uptimeStr = () => execFileSync('uptime', { encoding: 'utf8' }).trim();
async function waitLoad(label) {
  for (let i = 0; i < 40; i++) {
    const l = os.loadavg()[0];
    if (l <= 30) return { load1: r2(l), at: new Date().toISOString() };
    log(`load ${l.toFixed(1)} > 30, waiting before ${label}`);
    await sleep(15000);
  }
  return { load1: r2(os.loadavg()[0]), at: new Date().toISOString(), note: 'load stayed above 30 after 10 minutes' };
}
const loadNow = () => r2(os.loadavg()[0]);

// ------------------------------------------------------------------ build + serve the page
async function buildPage() {
  const vite = rq('vite');
  const src = path.join(TMP, 'page');
  fs.mkdirSync(src, { recursive: true });
  // The page imports the emulator by its package name and its build directory is outside the repository: the repository's node_modules are linked into it (a link from an earlier run is replaced), or the bare imports do not resolve.
  const link = path.join(src, 'node_modules');
  if (fs.lstatSync(link, { throwIfNoEntry: false })?.isSymbolicLink()) fs.unlinkSync(link);
  if (!fs.existsSync(link)) fs.symlinkSync(path.join(ROOT, 'node_modules'), link, 'dir');
  fs.copyFileSync(path.join(HERE, 'renderer-bench-page.txt'), path.join(src, 'main.js'));
  const manifest = JSON.parse(fs.readFileSync(path.join(ROOT, 'packages/kit/fonts/manifest.json'), 'utf8'));
  const mono = manifest.faces.find((f) => f.family === 'Int UI Mono');
  fs.copyFileSync(path.join(ROOT, 'packages/kit/fonts', mono.file), path.join(src, mono.file));
  fs.writeFileSync(path.join(src, 'index.html'), `<!doctype html><html><head><meta charset="utf-8"><title>t1 bench</title>
<style>
@font-face { font-family: 'Int UI Mono'; src: url('./${mono.file}') format('woff2'); font-style: normal; font-weight: 400; font-display: swap; unicode-range: ${mono.unicodeRange}; }
html, body { margin: 0; padding: 0; background: #191A1C; color: #BCBEC4; }
</style></head><body><div id="stage"></div><script type="module" src="./main.js"></script></body></html>`);
  fs.rmSync(path.join(TMP, 'dist'), { recursive: true, force: true });
  await vite.build({ root: src, configFile: false, logLevel: 'silent', base: './', build: { outDir: path.join(TMP, 'dist'), emptyOutDir: true, minify: 'esbuild', target: 'es2022', copyPublicDir: false, assetsInlineLimit: 0 } });
  fs.copyFileSync(path.join(src, mono.file), path.join(TMP, 'dist', mono.file));
  return fs.readdirSync(path.join(TMP, 'dist', 'assets')).map((f) => ({ f, bytes: fs.statSync(path.join(TMP, 'dist/assets', f)).size }));
}

function serve() {
  const dist = path.join(TMP, 'dist');
  const types = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.woff2': 'font/woff2' };
  const srv = http.createServer((req, res) => {
    const u = new URL(req.url, 'http://x');
    const p = path.join(dist, u.pathname === '/' ? 'index.html' : u.pathname);
    if (!p.startsWith(dist) || !fs.existsSync(p)) { res.writeHead(404); res.end(); return; }
    res.writeHead(200, { 'content-type': types[path.extname(p)] || 'application/octet-stream', 'cross-origin-opener-policy': 'same-origin', 'cross-origin-embedder-policy': 'require-corp', 'cross-origin-resource-policy': 'same-origin', 'cache-control': 'no-store' });
    fs.createReadStream(p).pipe(res);
  });
  return new Promise((resolve) => srv.listen(0, '127.0.0.1', () => resolve({ srv, url: `http://127.0.0.1:${srv.address().port}` })));
}

// ------------------------------------------------------------------ engines
async function launch(name) {
  if (name === 'webkit') return pw.webkit.launch();
  if (name === 'chromium') return pw.chromium.launch({ args: ['--font-render-hinting=none'] }); // = packages/kit/probes/probe-lib.mjs launch()
  if (name === 'chromium-gpu') return pw.chromium.launch({ channel: 'chromium', args: ['--font-render-hinting=none'] });
  throw new Error('engine ' + name);
}
const isChromium = (e) => e.startsWith('chromium');

const INSTRUMENT = () => {
  const c = (window.__c = { setTimeout: 0, setInterval: 0, raf: 0, rafMs: 0, sites: {}, trace: false });
  const site = () => (new Error().stack || '').split('\n').slice(2, 5).map((x) => x.trim().replace(/https?:\/\/[^/]+/g, '').slice(0, 100)).join(' < ');
  for (const n of ['setTimeout', 'setInterval']) {
    const o = window[n];
    window[n] = function (f, ...a) { c[n]++; if (c.trace) { const k = n + ' ' + site(); c.sites[k] = (c.sites[k] || 0) + 1; } return o.call(window, f, ...a); };
  }
  const ro = window.requestAnimationFrame;
  window.requestAnimationFrame = function (cb) {
    c.raf++;
    if (c.trace) { const k = 'raf ' + site(); c.sites[k] = (c.sites[k] || 0) + 1; }
    return ro.call(window, (t) => { const s = performance.now(); try { cb(t); } finally { c.rafMs += performance.now() - s; } });
  };
};

async function openPage(eng, ctxOpts = {}, query = '') {
  const ctx = await eng.browser.newContext({ viewport: { width: 1440, height: 900 }, ...ctxOpts });
  await ctx.addInitScript(INSTRUMENT);
  const page = await ctx.newPage();
  const consoleMsgs = [];
  const pageErrors = [];
  page.on('console', (m) => consoleMsgs.push({ type: m.type(), text: m.text().slice(0, 300) }));
  page.on('pageerror', (e) => pageErrors.push(String(e.message || e).slice(0, 300)));
  await page.goto(`${eng.url}/index.html?lh=${eng.lh ?? 1}${query}`);
  await page.waitForFunction(() => window.__benchReady === true);
  const p = { page, ctx, consoleMsgs, pageErrors, eng };
  p.close = async () => { try { await ctx.close(); } catch (e) { /* */ } };
  if (isChromium(eng.name)) p.cdp = await ctx.newCDPSession(page);
  return p;
}
const ev = (p, fn, arg) => p.page.evaluate(fn, arg);
const B = (p, name, ...args) => p.page.evaluate(({ name, args }) => window.bench[name](...args), { name, args });

// ------------------------------------------------------------------ processes
function psAll() {
  const out = execFileSync('ps', ['-axo', 'pid=,ppid=,rss=,command='], { encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 });
  return out.split('\n').map((l) => l.match(/^\s*(\d+)\s+(\d+)\s+(\d+)\s+(.*)$/)).filter(Boolean).map((m) => ({ pid: +m[1], ppid: +m[2], rssKB: +m[3], cmd: m[4] }));
}
const rssOf = (pid) => { const o = spawnSync('ps', ['-o', 'rss=', '-p', String(pid)], { encoding: 'utf8' }).stdout.trim(); return o ? Number(o) / 1024 : null; };
function descendants(all, root) { const out = []; const q = [root]; while (q.length) { const p = q.pop(); for (const x of all) if (x.ppid === p) { out.push(x); q.push(x.pid); } } return out; }

/** Attribute a renderer (Chromium) / WebContent (WebKit) process to a page by allocating a ballast in it and seeing whose RSS jumps. */
async function attribute(p) {
  const eng = p.eng;
  const all0 = psAll();
  let cands;
  if (isChromium(eng.name)) cands = descendants(all0, process.pid).filter((x) => /--type=renderer/.test(x.cmd));
  else cands = all0.filter((x) => /com\.apple\.WebKit\.WebContent/.test(x.cmd));
  const before = new Map(cands.map((c) => [c.pid, c.rssKB]));
  await B(p, 'allocBallast', 160);
  await sleep(400);
  const all1 = psAll();
  const jumps = cands.map((c) => { const n = all1.find((x) => x.pid === c.pid); return { pid: c.pid, delta: n ? (n.rssKB - before.get(c.pid)) / 1024 : -1 }; }).sort((a, b) => b.delta - a.delta);
  await B(p, 'freeBallast');
  await sleep(400);
  const best = jumps[0];
  const second = jumps[1];
  p.proc = { pid: best && best.delta > 120 ? best.pid : null, jumpMB: best ? r2(best.delta) : null, runnerUpMB: second ? r2(second.delta) : null, candidates: cands.length, method: isChromium(eng.name) ? 'descendant --type=renderer of this node pid, ballast-verified' : 'all com.apple.WebKit.WebContent pids, ballast-verified (XPC services have ppid 1, so no parent link)' };
  return p.proc;
}
function gpuRssMB(p) {
  const eng = p.eng;
  const all = psAll();
  if (isChromium(eng.name)) return descendants(all, process.pid).filter((x) => /--type=gpu-process/.test(x.cmd)).reduce((a, x) => a + x.rssKB / 1024, 0);
  return all.filter((x) => /com\.apple\.WebKit\.GPU/.test(x.cmd) && !eng.gpuBefore.has(x.pid)).reduce((a, x) => a + x.rssKB / 1024, 0);
}
function startPoll(p, ms = 100) {
  let peak = 0; let peakGpu = 0; let n = 0; let stop = false;
  const loop = (async () => {
    while (!stop) {
      if (p.proc?.pid) { const r = rssOf(p.proc.pid); if (r && r > peak) peak = r; n++; }
      await sleep(ms);
    }
  })();
  return async () => { stop = true; await loop; return { peakRssMB: r2(peak), samples: n }; };
}
async function snap(p, label) {
  const s = { label };
  if (isChromium(p.eng.name)) {
    await p.cdp.send('HeapProfiler.collectGarbage'); await p.cdp.send('HeapProfiler.collectGarbage');
    s.cdpHeapUsedMB = r2((await p.cdp.send('Runtime.getHeapUsage')).usedSize / 1e6);
    await p.cdp.send('Performance.enable');
    const m = (await p.cdp.send('Performance.getMetrics')).metrics;
    const g = (n) => m.find((x) => x.name === n)?.value;
    s.perfJSHeapUsedMB = r2(g('JSHeapUsedSize') / 1e6); s.perfJSHeapTotalMB = r2(g('JSHeapTotalSize') / 1e6);
    const ua = await B(p, 'uaMemory');
    if (ua.supported) s.uaMemoryMB = r2(ua.bytes / 1e6);
  } else s.gc = 'not forceable in WebKit (no CDP, no window.gc)';
  await sleep(150);
  if (p.proc?.pid) s.rendererRssMB = r2(rssOf(p.proc.pid));
  s.gpuProcRssMB = r2(gpuRssMB(p));
  return s;
}

// ------------------------------------------------------------------ calibration of lineHeight
async function calibrateLH(eng) {
  const p = await openPage(eng);
  const out = {};
  for (const r of RENDERERS) out[r] = await B(p, 'calibrate', r);
  await p.close();
  // the smallest lineHeight whose DOM row is 16 px high and whose css cell height is 16
  const pick = out.dom.table.find((x) => x.domRowH === 16 && x.cssH === 16);
  const webglSame = pick ? out.webgl.table.find((x) => x.lh === pick.lh) : null;
  eng.lh = pick ? pick.lh : null;
  return { chosen: eng.lh, dom: pick, webgl: webglSame, tables: out };
}

// ------------------------------------------------------------------ run engines
const Q = {};
async function forEngines(q, fn) {
  const { srv, url } = await serve();
  try {
    for (const name of ENGINES) {
      log(`${q} ${name}`);
      const eng = { name, url };
      try {
        eng.gpuBefore = new Set(psAll().filter((x) => /com\.apple\.WebKit\.GPU/.test(x.cmd)).map((x) => x.pid));
        eng.browser = await launch(name);
        eng.version = eng.browser.version();
        if (!eng.lhFromResult) {
          const cal = RESULT.lineHeight?.[name];
          if (cal?.chosen) eng.lh = cal.chosen; else { const c = await calibrateLH(eng); put('lineHeight', name, null, { ...c, tables: undefined, rawTables: c.tables }); }
        }
        await fn(eng);
      } catch (e) { log(`${q} ${name} FAILED`, e.stack || e); put(q, name, null, { error: String(e.stack || e).slice(0, 800) }); }
      try { await eng.browser.close(); } catch (e) { /* */ }
    }
  } finally { srv.close(); }
}

// ================================================================== B2 construction
Q.B2 = async () => {
  await forEngines('B2', async (eng) => {
    const coldReps = REPS ?? 5; const warmReps = Math.max(2, Math.ceil(coldReps / 2));
    for (const r of RENDERERS) {
      const w = await waitLoad('B2');
      const cold = []; const warm = [];
      for (let i = 0; i < coldReps; i++) { const p = await openPage(eng); cold.push((await B(p, 'construct', r, 1)).first); await p.close(); }
      for (let i = 0; i < warmReps; i++) { const p = await openPage(eng); warm.push((await B(p, 'construct', r, 10)).nth); await p.close(); }
      const agg = (rows) => Object.fromEntries(Object.keys(rows[0]).map((k) => [k, { median: r2(median(rows.map((x) => x[k]))), min: r2(Math.min(...rows.map((x) => x[k]))), max: r2(Math.max(...rows.map((x) => x[k]))) }]));
      put('B2', eng.name, r, { cold: agg(cold), warm: agg(warm), coldRuns: cold.length, warmRuns: warm.length, load1: w.load1, load1After: loadNow(), version: eng.version, lineHeight: eng.lh });
      log('B2', eng.name, r, 'cold total', agg(cold).firstFrameTotal.median, 'warm', agg(warm).firstFrameTotal.median);
    }
  });
};

  const l = [
    { kind: 'W1', mode: 'chunk', pace: 'free', reps: 5 }, { kind: 'W1', mode: 'raf', pace: 'free', reps: 5 },

    const values = ['#191A1C', '#BCBEC4', 'rgb(25 26 28)', 'rgb(25, 26, 28)', 'rgba(33, 66, 131, 0.5)', '#21428380', 'hsl(220 6% 10%)', 'oklch(0.2 0.005 260)', 'oklch(0.7 0.15 30)', 'color-mix(in srgb, #ff0000 50%, #0000ff)', 'color-mix(in oklab, #191A1C 80%, white)', 'color(display-p3 1 0 0)', 'color(srgb 0.1 0.2 0.3)', 'light-dark(#fff, #000)', 'transparent', 'currentcolor', 'red'];
    out.resolution = await B(p0, 'resolveColors', values);
    // custom-property chain: --a: var(--b), --b: #hex
    out.chain = await p0.page.evaluate(() => { const el = document.createElement('div'); document.body.appendChild(el); el.style.cssText = '--b:#BCBEC4;--a:var(--b);color:var(--a)'; const v = getComputedStyle(el).color; el.remove(); return v; });
    out.registeredProperty = await p0.page.evaluate(() => { try { CSS.registerProperty({ name: '--reg', syntax: '<color>', inherits: true, initialValue: 'red' }); const el = document.createElement('div'); document.body.appendChild(el); el.style.setProperty('--reg', '#191A1C'); const v = getComputedStyle(el).getPropertyValue('--reg'); el.remove(); return v; } catch (e) { return 'ERR ' + e.message; } });
    out.rawCustomPropertyValue = await p0.page.evaluate(() => { const el = document.createElement('div'); document.body.appendChild(el); el.style.setProperty('--x', ' #191A1C '); const v = getComputedStyle(el).getPropertyValue('--x'); el.remove(); return JSON.stringify(v); });
    await p0.close();
    for (const r of RENDERERS) {
      const w = await waitLoad('B10');
      const rr = {};
      const p = await openPage(eng);
      const hex = await p.page.evaluate(() => window.bench.themeFrom());
      const viaComputed = await p.page.evaluate((t) => window.bench.themeViaComputed(t), hex);
      const spaceSyntax = Object.fromEntries(Object.entries(viaComputed).map(([k, v]) => [k, v.replace(/rgba?\((\d+), (\d+), (\d+)\)/, 'rgb($1 $2 $3)')]));
      const negative = { ...viaComputed, red: 'rgb(247, 84, 101)' }; // one level off in blue (100 -> 101)
      const alphaHex = { ...hex, selectionBackground: '#21428380' };
      const alphaComputed = { ...viaComputed, selectionBackground: await p.page.evaluate(() => window.bench.themeViaComputed({ x: '#21428380' }).x) };
      const oklchBg = { ...hex, background: 'oklch(0.2 0.005 260)' };
      const oklchBgResolved = { ...hex, background: await p.page.evaluate(() => window.bench.themeViaComputed({ x: 'oklch(0.2 0.005 260)' }).x) };
      rr.sampleValues = { viaComputed: { background: viaComputed.background, foreground: viaComputed.foreground, red: viaComputed.red, selectionBackground: viaComputed.selectionBackground }, space: spaceSyntax.red, alphaComputedSelection: alphaComputed.selectionBackground, oklchBgResolved: oklchBgResolved.background };
      const shot = async (theme, name) => { const info = await p.page.evaluate((a) => window.bench.themePattern(a), { renderer: r, theme }); const buf = await png(p, info.rect, name ? `B10-${r}-${name}` : null); return { info, buf }; };
      const A = await shot(hex, 'hex');
      const Bc = await shot(viaComputed, 'computed-rgb');
      const Cs = await shot(spaceSyntax, 'space-rgb');
      const Neg = await shot(negative, 'negative-control');
      const Ah = await shot(alphaHex, 'alpha-hex8');
      const Ac = await shot(alphaComputed, 'alpha-computed-rgba');
      const Ob = await shot(oklchBg, 'raw-oklch-bg');
      const Or = await shot(oklchBgResolved, 'resolved-oklch-bg');
      const cell = A.info.cell;
      const pts = [];
      for (let i = 0; i < 16; i++) pts.push([Math.round((25) * cell.cssW), Math.round((i + 1) * 16 + 8)]);
      pts.push([600, 8 * 16 + 2]);
      const dA = await diffPng(p, A.buf, Bc.buf, pts);
      rr.cell = cell; rr.colsRows = [A.info.cols, A.info.rows]; rr.hasFocus = A.info.hasFocus;
      rr.hex_vs_computedRgb = (({ differingPixels, maxChannelDelta, nonBackgroundPixelsA }) => ({ differingPixels, maxChannelDelta, nonBackgroundPixelsA }))(dA);
      rr.hex_vs_spaceRgb = await diffPng(p, A.buf, Cs.buf);
      rr.hex_vs_negativeControl = await diffPng(p, A.buf, Neg.buf);
      rr.alphaHex8_vs_alphaComputedRgba = await diffPng(p, Ah.buf, Ac.buf);
      rr.alphaHex8_vs_opaqueSelection = await diffPng(p, Ah.buf, A.buf);
      rr.rawOklchBg_vs_computedStyleOfOklchBg = await diffPng(p, Ob.buf, Or.buf);
      rr.rawOklchBg_vs_hexBg = await diffPng(p, Ob.buf, A.buf);
      const bgPt = [[600, 8 * 16 + 2]];
      rr.rawOklchBg_backgroundPixel = (await p.page.evaluate((x) => window.bench.pixelsOf(x), { a: Ob.buf, points: bgPt })).points[0]; // 21,22,24 = what a canvas makes of that oklch(); 0,0,0 = xterm ignored it
      // absolute pixels: the ANSI background blocks and the default background, hex theme
      const expected = p.page.evaluate(() => window.bench.ANSI_HEX);
      const exp = (await expected).map((h) => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)]);
      exp.push([25, 26, 28]);
      rr.absolutePixels = dA.points.map((got, i) => ({ i, expected: exp[i], got, equal: got.every((v, k) => v === exp[i][k]) }));
      rr.absoluteMismatches = rr.absolutePixels.filter((x) => !x.equal).length;
      rr.load1 = w.load1;
      out[r] = rr;
      await p.close();
      log('B10', eng.name, r, 'hex vs computed diff', rr.hex_vs_computedRgb.differingPixels, 'neg', rr.hex_vs_negativeControl.differingPixels, 'abs mismatches', rr.absoluteMismatches);
    }
    put('B10', eng.name, null, out);
  });
};

// ================================================================== main
async function main() {
  if (argv.includes('--list')) { console.log(Object.keys(Q).join(' ')); return; }
  const load = await waitLoad('start');
  const meta = { started: new Date().toISOString(), uptime: uptimeStr(), load1AtStart: load.load1, node: process.version, host: `${os.type()} ${os.release()} ${os.arch()}, ${os.cpus().length} cpus`, engines: ENGINES, playwright: JSON.parse(fs.readFileSync(path.join(ROOT, 'node_modules/playwright/package.json'), 'utf8')).version };
  const files = await buildPage();
  meta.pageBuild = files;
  const list = ONLY ?? Object.keys(Q);
  save('meta', { ...(RESULT.meta ?? {}), [meta.started]: { ...meta, only: list } });
  for (const q of list) {
    if (!Q[q]) { log('unknown question', q); continue; }
    log('=== ', q, 'load', loadNow());
    await Q[q]();
    const m = RESULT.meta; m[meta.started][q + 'Done'] = { at: new Date().toISOString(), uptime: uptimeStr() }; save('meta', m);
  }
  log('done');
}
main().then(() => process.exit(0), (e) => { console.error(e); process.exit(1); });
