const now = () => performance.now();
const raf = () => new Promise((r) => requestAnimationFrame(r));
const raf2 = async () => { await raf(); await raf(); };
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const r2 = (n) => Math.round(n * 100) / 100;

function pct(sorted, p) {
  if (!sorted.length) return null;
  const i = Math.min(sorted.length - 1, Math.max(0, Math.ceil((p / 100) * sorted.length) - 1));
  return sorted[i];
}
function stats(arr) {
  const s = [...arr].sort((a, b) => a - b);
  return { n: s.length, min: r2(s[0] ?? NaN), p50: r2(pct(s, 50) ?? NaN), p95: r2(pct(s, 95) ?? NaN), p99: r2(pct(s, 99) ?? NaN), max: r2(s[s.length - 1] ?? NaN), mean: r2(s.reduce((a, b) => a + b, 0) / (s.length || 1)) };
}

function baseOpts(o = {}) {
  return { fontFamily: FONT, fontSize: 12, lineHeight: LH, cursorBlink: false, scrollback: 5000, allowProposedApi: false, theme: THEME, ...o };
}

function makeHost(w = HOST_W, h = HOST_H, parent = document.getElementById('stage'), css = '') {
  const host = document.createElement('div');
  host.style.cssText = `width:${w}px;height:${h}px;overflow:hidden;position:relative;${css}`;
  parent.appendChild(host);
  return host;
}

async function fontReady() {
  await document.fonts.load('12px "Int UI Mono"');
}

/** Cell size the renderer settled on (private API, bench only). */
function cellSize(term) {
  const d = term._core._renderService.dimensions;
  return { cssW: d.css.cell.width, cssH: d.css.cell.height, devW: d.device.cell.width, devH: d.device.cell.height };
}

/** Sum the time xterm spends in its own parse slices (WriteBuffer._innerWrite); bench only. */
function instrumentParse(term) {
  const wb = term._core._writeBuffer;
  const orig = wb._innerWrite.bind(wb);
  const st = { ms: 0, calls: 0, maxSlice: 0 };
  wb._innerWrite = (...a) => {
    const s = now();
    try { return orig(...a); } finally { const d = now() - s; st.ms += d; st.calls += 1; if (d > st.maxSlice) st.maxSlice = d; }
  };
  return st;
}

async function create({ renderer = 'dom', scrollback = 5000, cursorBlink = false, host, hostW = HOST_W, hostH = HOST_H, theme, opts = {}, fit = true, parse = false } = {}) {
  await fontReady();
  host = host || makeHost(hostW, hostH);
  const term = new Terminal(baseOpts({ scrollback, cursorBlink, ...(theme ? { theme } : {}), ...opts }));
  term.open(host);
  const fitA = new FitAddon();
  term.loadAddon(fitA);
  if (fit) fitA.fit();
  let webgl = null;
  if (renderer === 'webgl') {
    webgl = new WebglAddon();
    term.loadAddon(webgl);
  }
  const ps = parse ? instrumentParse(term) : null;
  return { term, fitA, webgl, host, ps };
}

const live = { cur: null };

function teardown(t) {
  if (!t) return;
  try { t.term.dispose(); } catch (e) { /* already disposed */ }
  t.host.remove();
}

// ---------------------------------------------------------------- streams

const LINE80 = ('The quick brown fox jumps over the lazy dog 0123456789 ~!@#$%^&*()_+-=[]{}|;:,.<>/? abcdefghijklmnop').slice(0, 80);
const MARK = 'LAST-LINE-OK';
const enc = new TextEncoder();

function makeStream(kind, { reuse = false } = {}) {
  const line = kind === 'W1' ? 'y' : LINE80;
  const nLines = kind === 'W1' ? 200000 : kind === 'W2' ? 1000000 : kind.startsWith('B') ? Number(kind.slice(1)) : 0;
  const lineBytes = enc.encode(line + '\r\n');
  const L = lineBytes.length;
  const tail = enc.encode(MARK + '\r\n');
  const bodyLen = L * nLines;
  const total = bodyLen + tail.length;
  const base = new Uint8Array(CHUNK + L);
  for (let i = 0; i < base.length; i += L) base.set(lineBytes.subarray(0, Math.min(L, base.length - i)), i);
  const chunks = Math.ceil(total / CHUNK);
  const cache = new Map();
  const gen = (i) => {
    const s = i * CHUNK;
    const e = Math.min(total, s + CHUNK);
    if (e <= bodyLen && reuse && e - s === CHUNK) { // memory runs: no per-chunk garbage, the same few arrays are written over and over
      let a = cache.get(s % L);
      if (!a) { a = base.slice(s % L, (s % L) + CHUNK); cache.set(s % L, a); }
      return a;
    }
    if (e <= bodyLen) return base.slice(s % L, (s % L) + (e - s));
    const out = new Uint8Array(e - s);
    if (s < bodyLen) out.set(base.subarray(s % L, (s % L) + (bodyLen - s)), 0);
    const from = Math.max(s, bodyLen);
    out.set(tail.subarray(from - bodyLen, e - bodyLen), from - s);
    return out;
  };
  return { gen, chunks, total, lineBytes: L, nLines, line };
}

/** Frame sampler: rAF timestamps. */
function frameSampler() {
  const ts = [];
  let running = true;
  const loop = (t) => { ts.push(t); if (running) requestAnimationFrame(loop); };
  requestAnimationFrame(loop);
  return {
    stop() { running = false; return ts; },
    intervals() { const a = []; for (let i = 1; i < ts.length; i++) a.push(ts[i] - ts[i - 1]); return a; },
    count() { return ts.length; },
  };
}

function longTaskObserver() {
  const out = { supported: false, count: 0, totalMs: 0, maxMs: 0, durations: [] };
  try {
    if (!(PerformanceObserver.supportedEntryTypes || []).includes('longtask')) return { obs: null, out };
    out.supported = true;
    const obs = new PerformanceObserver((l) => { for (const e of l.getEntries()) { out.count++; out.totalMs += e.duration; out.durations.push(e.duration); if (e.duration > out.maxMs) out.maxMs = e.duration; } });
    obs.observe({ type: 'longtask' });
    return { obs, out };
  } catch (e) { return { obs: null, out }; }
}

function counters() { const c = window.__c; return { st: c.setTimeout, si: c.setInterval, raf: c.raf, rafMs: c.rafMs }; }
function counterDelta(a, b) { return { setTimeout: b.st - a.st, setInterval: b.si - a.si, raf: b.raf - a.raf, rafMs: r2(b.rafMs - a.rafMs) }; }

function mutationCounter() {
  const m = { records: 0, byType: {}, targets: {} };
  const mo = new MutationObserver((list) => {
    for (const r of list) {
      m.records++;
      m.byType[r.type] = (m.byType[r.type] || 0) + 1;
      const t = r.target;
      const k = (t.nodeType === 1 ? t.tagName.toLowerCase() + (t.className && typeof t.className === 'string' ? '.' + t.className.split(' ')[0] : '') : '#' + t.nodeName) + (r.attributeName ? '[' + r.attributeName + ']' : '');
      m.targets[k] = (m.targets[k] || 0) + 1;
    }
  });
  mo.observe(document, { subtree: true, childList: true, attributes: true, characterData: true });
  return { m, stop() { const rest = mo.takeRecords(); for (const r of rest) { m.records++; m.byType[r.type] = (m.byType[r.type] || 0) + 1; } mo.disconnect(); return m; } };
}

/**
 * Pump a stream into a terminal.
 *  mode: 'chunk' (one write per chunk) | 'raf' (everything that arrived since the last frame, concatenated, one write per rAF)
 *  pace: 'free' (one chunk per MessageChannel macrotask, as fast as the page allows) | 'bp' (same, but at most capBytes written and not yet parsed)
 *        | 'burst' (every chunk written in one synchronous loop)
 */
async function pump(term, stream, { mode, pace, capBytes = 2 * 1024 * 1024 }) {
  const res = { accepted: 0, discarded: 0, discardedBytes: 0, writes: 0, firstError: null, gaps: [] };
  let unacked = 0;
  const pending = [];
  let pendingLen = 0;
  let produced = 0;
  let producerDone = false;
  let rafOn = false;
  const t0 = now();
  const doWrite = (data, ack) => {
    res.writes++;
    unacked += data.length;
    try {
      term.write(data, () => { unacked -= data.length; if (ack) ack(); });
      res.accepted++;
    } catch (e) {
      res.discarded++; res.discardedBytes += data.length; unacked -= data.length;
      if (!res.firstError) res.firstError = String(e && e.message || e);
    }
  };
  const mc = new MessageChannel();
  let posted = false;
  let resume = null;
  let lastStart = 0;
  const post = () => { if (!posted) { posted = true; mc.port2.postMessage(0); } };
  const flushRaf = () => {
    if (!pending.length) return;
    let big;
    if (pending.length === 1) big = pending[0];
    else { big = new Uint8Array(pendingLen); let o = 0; for (const c of pending) { big.set(c, o); o += c.length; } }
    pending.length = 0; pendingLen = 0;
    doWrite(big, resume);
  };
  const rafLoop = () => {
    flushRaf();
    if (!producerDone || pending.length) requestAnimationFrame(rafLoop); else rafOn = false;
  };
  const canSend = () => pace !== 'bp' || (unacked + pendingLen) < capBytes;
  const deliver = (chunk) => {
    if (mode === 'chunk') doWrite(chunk, resume);
    else { pending.push(chunk); pendingLen += chunk.length; if (!rafOn) { rafOn = true; requestAnimationFrame(rafLoop); } }
  };
  const done = new Promise((resolve) => {
    const finish = () => { producerDone = true; resolve(); };
    if (pace === 'burst') {
      for (; produced < stream.chunks; produced++) deliver(stream.gen(produced));
      finish();
      return;
    }
    resume = () => { if (!producerDone && produced < stream.chunks && !posted && canSend()) post(); };
    mc.port1.onmessage = () => {
      posted = false;
      const s = now();
      if (lastStart) res.gaps.push(s - lastStart);
      lastStart = s;
      deliver(stream.gen(produced++));
      if (produced >= stream.chunks) { finish(); return; }
      if (canSend()) post();
    };
    post();
  });
  res.firstWriteAt = t0;
  await done;
  mc.port1.onmessage = null; mc.port1.close(); // a live port with a handler keeps the terminal reachable
  res.producedAt = now();
  // a final sentinel write: its callback fires after every earlier byte is parsed (retry while xterm refuses data)
  while (pending.length) await raf();
  let sentinelAt = null;
  for (;;) {
    try { await new Promise((resolve) => term.write('', resolve)); sentinelAt = now(); break; } catch (e) { await sleep(10); }
  }
  res.drainEnd = sentinelAt;
  res.drainMs = r2(sentinelAt - t0);
  res.produceMs = r2(res.producedAt - t0);
  return res;
}

function lastLineCheck(t) {
  const term = t.term;
  const b = term.buffer.active;
  const lineAt = (y) => b.getLine(y)?.translateToString(true) ?? null;
  const last = lineAt(b.baseY + b.cursorY - 1);
  const out = { bufferLength: b.length, baseY: b.baseY, viewportY: b.viewportY, cursorY: b.cursorY, lastBufferLine: last, lastLineOk: last === MARK, atBottom: b.viewportY === b.baseY, rows: term.rows, cols: term.cols };
  const rowsEl = t.host.querySelector('.xterm-rows');
  if (rowsEl) {
    const kids = [...rowsEl.children];
    const txt = kids.map((k) => k.textContent.replace(/\u00a0/g, ' ').trimEnd());
    out.domLastRendered = txt.filter((x) => x.length).slice(-1)[0] ?? null;
    out.domRenderedOk = out.domLastRendered === MARK;
  }
  return out;
}

async function calibrate(renderer) {
  await fontReady();
  const out = [];
  for (const lh of [1, 1.05, 1.1, 1.15, 1.2, 1.25, 1.3, 1.35, 1.4]) {
    const host = makeHost();
    const term = new Terminal(baseOpts({ lineHeight: lh }));
    term.open(host);
    const f = new FitAddon();
    term.loadAddon(f);
    let w = null;
    if (renderer === 'webgl') { w = new WebglAddon(); term.loadAddon(w); }
    f.fit();
    term.write('hello');
    await raf2();
    await raf2();
    const rowEl = host.querySelector('.xterm-rows > div');
    const c = cellSize(term);
    out.push({ lh, ...c, domRowH: rowEl ? rowEl.getBoundingClientRect().height : null, cols: term.cols, rows: term.rows });
    term.dispose();
    host.remove();
  }
  return { renderer, table: out, fontLoaded: document.fonts.check('12px "Int UI Mono"'), dpr: devicePixelRatio };
}

// ---------------------------------------------------------------- B2 construction

async function construct(renderer, nth = 1) {
  await fontReady();
  const res = [];
  let last = null;
  for (let i = 1; i <= nth; i++) {
    teardown(last); last = null;
    await raf2();
    const host = makeHost();
    const T = {};
    const renders = [];
    const t0 = now();
    const term = new Terminal(baseOpts()); T.new = now() - t0;
    let s = now();
    term.open(host); T.open = now() - s;
    const rdr = term.onRender(() => renders.push(now()));
    s = now();
    const fit = new FitAddon(); term.loadAddon(fit); T.loadFit = now() - s;
    s = now();
    fit.fit(); T.fit = now() - s;
    let w = null;
    if (renderer === 'webgl') {
      s = now(); w = new WebglAddon(); term.loadAddon(w); T.loadWebgl = now() - s;
      T.colsBeforeRefit = term.cols; T.cellWidthAfterWebgl = cellSize(term).cssW;
      s = now(); fit.fit(); T.refitAfterWebgl = now() - s; // the WebGL renderer rounds the cell to whole device pixels: the grid may change
    }
    const tEnd = now();
    T.sync = tEnd - t0;
    const nR = renders.length;
    await new Promise((resolve) => term.write('first line of output\r\n', resolve));
    const tCb = now();
    if (renders.length === nR) await new Promise((r) => { const d = term.onRender(() => { d.dispose(); r(); }); });
    const tRender = now();
    await raf();
    const tPaint = now();
    T.writeCb = tCb - tEnd; T.render = tRender - tEnd; T.paint = tPaint - tEnd;
    T.firstFrameTotal = tPaint - t0;
    T.firstOnRenderSinceOpen = renders.length ? renders[0] - (t0 + T.new) : null;
    T.cols = term.cols; T.rows = term.rows;
    rdr.dispose();
    for (const k of Object.keys(T)) if (typeof T[k] === 'number') T[k] = r2(T[k]);
    res.push(T);
    last = { term, host };
  }
  teardown(last);
  return { nth: res[nth - 1], first: res[0], all: res };
}

// ---------------------------------------------------------------- B3 flood

async function flood(cfg) {
  const { renderer, kind, mode, pace, scrollback = 5000, capBytes } = cfg;
  const t = await create({ renderer, scrollback, parse: true });
  live.cur = t;
  const { term } = t;
  await raf2(); await sleep(150); await raf2();
  const base = frameSampler();
  await sleep(500);
  base.stop();
  const baseline = stats(base.intervals());
  const stream = makeStream(kind);
  const { obs, out: lt } = longTaskObserver();
  const c0 = counters();
  const mut = mutationCounter();
  const fs = frameSampler();
  const g = await pump(term, stream, { mode, pace, capBytes });
  const c1 = counters();
  const frameTs = fs.stop();
  const ivs = fs.intervals();
  await raf2();
  const muts = mut.stop();
  if (obs) { await sleep(120); obs.disconnect(); }
  const check = lastLineCheck(t);
  const out = {
    cfg: { ...cfg, scrollback },
    stream: { chunks: stream.chunks, bytes: stream.total, lines: stream.nLines },
    baselineFrameInterval: baseline,
    frames: frameTs.length,
    frameInterval: stats(ivs),
    intervals: ivs.map(r2),
    framesOver50ms: ivs.filter((x) => x > 50).length,
    drainMs: g.drainMs, produceMs: g.produceMs,
    throughputMBs: r2(stream.total / 1e6 / (g.drainMs / 1000)),
    accepted: g.accepted, discarded: g.discarded, discardedMB: r2(g.discardedBytes / 1e6), firstError: g.firstError,
    writes: g.writes,
    parseMs: r2(t.ps.ms), parseSlices: t.ps.calls, maxParseSliceMs: r2(t.ps.maxSlice),
    loopGap: stats(g.gaps),
    longTasks: { supported: lt.supported, count: lt.count, totalMs: r2(lt.totalMs), maxMs: r2(lt.maxMs), over100: lt.durations.filter((x) => x > 100).length },
    counters: counterDelta(c0, c1),
    mutations: muts.records,
    check,
    heapUsedMB: performance.memory ? r2(performance.memory.usedJSHeapSize / 1e6) : null,
  };
  return out;
}

async function floodDispose() { teardown(live.cur); live.cur = null; await raf2(); return true; }

const lossState = { terms: [], events: [], refDom: null };
async function ctxLimit({ max = 40, wait = 120 }) {
  await fontReady();
  const stage = document.getElementById('stage');
  stage.style.cssText = 'display:flex;flex-wrap:wrap;';
  const ev = [];
  const T0 = now();
  const created = [];
  for (let i = 0; i < max; i++) {
    const host = makeHost(240, 64, stage);
    const term = new Terminal(baseOpts({ scrollback: 100 }));
    term.open(host);
    let w = null; let err = null;
    try { w = new WebglAddon(); term.loadAddon(w); } catch (e) { err = String(e && e.message || e); }
    const rec = { idx: i, term, host, w, err, lostAt: null, addonLossAt: null, restoredAt: null };
    if (w) {
      for (const cv of host.querySelectorAll('canvas')) { // xterm adds more than one canvas; the event fires on the WebGL one only
        cv.addEventListener('webglcontextlost', () => { rec.lostAt = now() - T0; ev.push({ type: 'webglcontextlost (canvas event)', idx: i, createdSoFar: created.length, aliveAtThatTime: created.filter((c) => c.w && !c.disposedAt && c.lostAt === null).length, t: r2(now() - T0) }); });
        cv.addEventListener('webglcontextrestored', () => { rec.restoredAt = now() - T0; ev.push({ type: 'restored', idx: i, t: r2(now() - T0) }); });
      }
      w.onContextLoss(() => { rec.addonLossAt = now() - T0; ev.push({ type: 'addon.onContextLoss', idx: i, t: r2(now() - T0) }); });
    }
    term.write(`terminal ${i}\r\nline two`);
    created.push(rec);
    await sleep(wait);
    if (err) { ev.push({ type: 'addonError', idx: i, err }); break; }
  }
  lossState.terms = created;
  lossState.events = ev;
  await sleep(3600); // let xterm's 3 s restoration timer fire onContextLoss
  const lost = created.filter((c) => c.lostAt !== null).map((c) => c.idx);
  return { created: created.length, events: ev, lostIdx: lost, addonLossIdx: created.filter((c) => c.addonLossAt !== null).map((c) => c.idx), restoredIdx: created.filter((c) => c.restoredAt !== null).map((c) => c.idx) };
}
function lossInspect(idx) {
  const c = lossState.terms[idx];
  const canv = [...c.host.querySelectorAll('canvas')].map((x) => ({ cls: x.className, w: x.width, h: x.height, display: getComputedStyle(x).display }));
  const rows = c.host.querySelector('.xterm-rows');
  return { canvases: canv, rowsDisplay: rows ? getComputedStyle(rows).display : null, rowsText: rows ? [...rows.children].map((k) => k.textContent.trim()).filter(Boolean).slice(0, 4) : null };
}
async function lossWrite(idx) { const c = lossState.terms[idx]; c.term.write('\r\nAFTER-LOSS-WRITE'); await raf2(); await sleep(100); return lossInspect(idx); }
async function lossDispose(idx) { const c = lossState.terms[idx]; c.w.dispose(); c.disposedAt = now(); await raf2(); await sleep(100); return lossInspect(idx); }

const togState = { t: null };
function fillContent(term, lines) {
  let s = '';
  for (let i = 0; i < lines; i++) s += `\x1b[3${i % 8}mline ${String(i).padStart(5, '0')}\x1b[0m the quick brown fox \x1b[1;4${(i + 3) % 8}m jumps \x1b[0m over ${i * 7919 % 100003}\r\n`;
  return new Promise((r) => term.write(s, r));
}
async function togSetup({ lines = 6000, hide = 'visible' }) {
  teardown(togState.t);
  const t = await create({ renderer: 'dom', scrollback: 5000 });
  await fillContent(t.term, lines);
  t.term.scrollToBottom();

/** resolve each theme value to the string getComputedStyle gives (what the product would pass) */
function themeViaComputed(theme) {
  const keys = Object.keys(theme);
  const res = resolveColors(keys.map((k) => theme[k]));
  return Object.fromEntries(keys.map((k, i) => [k, res[i].computed]));
}

/** pixel diff between two PNGs (base64); returns counts of differing pixels and, optionally, the pixel at points of the first */
function comparePng({ a, c, points = [] }) {
  const load = async (b64) => {
    const img = new Image();
    img.src = `data:image/png;base64,${b64}`;
    await img.decode();
    const cv = document.createElement('canvas');
    cv.width = img.width; cv.height = img.height;
    const ctx = cv.getContext('2d');
    ctx.drawImage(img, 0, 0);
    return ctx.getImageData(0, 0, img.width, img.height);
  };
  return Promise.all([load(a), load(c)]).then(([x, y]) => {
    if (x.width !== y.width || x.height !== y.height) return { sizeMismatch: [x.width, x.height, y.width, y.height] };
    let diff = 0; let maxD = 0; let nonBg = 0;
    const bg = [x.data[0], x.data[1], x.data[2]];
    for (let i = 0; i < x.data.length; i += 4) {
      const d = Math.max(Math.abs(x.data[i] - y.data[i]), Math.abs(x.data[i + 1] - y.data[i + 1]), Math.abs(x.data[i + 2] - y.data[i + 2]));
      if (d > 0) { diff++; if (d > maxD) maxD = d; }
      if (x.data[i] !== bg[0] || x.data[i + 1] !== bg[1] || x.data[i + 2] !== bg[2]) nonBg++;
    }
    const px = points.map(([px0, py0]) => { const i = (py0 * x.width + px0) * 4; return [x.data[i], x.data[i + 1], x.data[i + 2]]; });
    return { width: x.width, height: x.height, differingPixels: diff, maxChannelDelta: maxD, fractionDiffering: Math.round((diff / (x.width * x.height)) * 1e6) / 1e6, nonBackgroundPixelsA: nonBg, points: px };
  });
}
function pixelsOf({ a, points }) {
  return (async () => {
    const img = new Image(); img.src = `data:image/png;base64,${a}`; await img.decode();
    const cv = document.createElement('canvas'); cv.width = img.width; cv.height = img.height;
    const ctx = cv.getContext('2d'); ctx.drawImage(img, 0, 0);
    const d = ctx.getImageData(0, 0, img.width, img.height).data;
    return { width: img.width, height: img.height, points: points.map(([x, y]) => { const i = (y * img.width + x) * 4; return [d[i], d[i + 1], d[i + 2]]; }) };
  })();
}
