# substitutes for the real one, so the repository is never edited. A control is CAUGHT when the tests named for it fail (a fault that
# does not build is reported as BUILD and counts for nothing). Four run at a time.
# Run from the repository root (drop the .txt):  python3 mutants.py [substring of a fault name ...]

# A fault whose control is the race detector runs the tests with -race.
RACE = ("M19",)
only = sys.argv[1:]
env = dict(os.environ, GOWORK="off", GOTOOLCHAIN="local")


def run(name, which, old, new):
    assert old in sources[which], name + ": the text to change is not in the file"
    assert sources[which].count(old) == 1, name + ": the text to change is not unique"
    with tempfile.TemporaryDirectory() as scratch:
        mutant = os.path.join(scratch, files[which])
        open(mutant, "w").write(sources[which].replace(old, new, 1))
        overlay = os.path.join(scratch, "overlay.json")
        json.dump({"Replace": {pkg + files[which]: mutant}}, open(overlay, "w"))
        command = ["go", "test", "-overlay", overlay, "./backend/pkg/plugin/resource/", "-run", "TestConnectionMetadataLastKnown|TestLastKnown", "-count=1", "-timeout", "150s"]
        done = subprocess.run(command + (["-race"] if name.startswith(RACE) else []), cwd=root, env=env, capture_output=True, text=True)
    out = done.stdout + done.stderr
    names = re.findall(r"--- FAIL: (\S+)", out)
    failed = sorted({m for m in names if "/" not in m}) or sorted(set(names))[:2] or re.findall(r"panic: [^\n]{0,60}", out)[:1]
    if "[build failed]" in out:
        return name, "BUILD   ", failed
    return name, "CAUGHT  " if done.returncode != 0 else "SURVIVED", failed


selected = [(n, *f) for n, f in faults.items() if not only or any(o in n for o in only)]
counts = {"CAUGHT  ": 0, "SURVIVED": 0, "BUILD   ": 0}
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    for name, status, failed in pool.map(lambda args: run(*args), selected):
        counts[status] += 1
        print(status, name, failed, flush=True)
print("%d of %d caught, %d survived, %d did not build" % (counts["CAUGHT  "], len(selected), counts["SURVIVED"], counts["BUILD   "]))
