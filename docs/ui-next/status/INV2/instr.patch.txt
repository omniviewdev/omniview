+package resource
+
+import "sync/atomic"
+
+var debugRetireCalls atomic.Int64
+
+// DebugSourceCounts is an INV2 investigation probe (never committed).
+func (c *controller) DebugSourceCounts() map[string]int {
+	sc := c.sources
+	sc.mu.Lock()
+	defer sc.mu.Unlock()
+	return map[string]int{
+		"sourceSessions": len(sc.sessions), "sourceViews": len(sc.views), "sourceInterests": len(sc.interests),
+		"sourceViewInterests": len(sc.viewInterests), "sourceConnInterests": len(sc.connectionInterests),
+		"retiredOwners": len(sc.retiredOwners), "retireCalls": int(debugRetireCalls.Load()),
+	}
+}

+package main
+
+import (
+	"encoding/json"
+	"os"
+	"time"
+
+	"github.com/omniviewdev/omniview/backend/pkg/invocation"
+)
+
+// inv2Counters writes the realm's and the source coordinator's counts to the file named by INV2_COUNTERS every 250 ms. INV2 investigation probe (never committed).
+func inv2Counters(realm *invocation.WindowRealm, ctrl any) {
+	path := os.Getenv("INV2_COUNTERS")
+	if path == "" {
+		return
+	}
+	for range time.Tick(250 * time.Millisecond) {
+		out := realm.DebugCounts()
+		if c, ok := ctrl.(interface{ DebugSourceCounts() map[string]int }); ok {
+			for k, v := range c.DebugSourceCounts() {
+				out[k] = v
+			}
+		}
+		data, _ := json.Marshal(out)
+		_ = os.WriteFile(path+".tmp", data, 0o600)
+		_ = os.Rename(path+".tmp", path)
+	}
+}
