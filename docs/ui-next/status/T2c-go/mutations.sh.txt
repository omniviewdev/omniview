#!/bin/bash
# Negative controls for T2c-go: each fault is put into a SCRATCH copy of one Go source file and built with
# `go test -overlay`; the worktree is never edited. A control passes when the test named for it turns red.
# The two controls of the TypeScript side swap the file in place, run vitest, and put the file back (checked with cmp).
# ROOT is the worktree (default: the current directory); ONLY=<regexp> runs the controls whose name matches.
export GOWORK=off GOTOOLCHAIN=local
WT=${ROOT:-$PWD}
M=${MDIR:-/tmp/t2c-mut}
ONLY=${ONLY:-.}
rm -rf "$M" && mkdir -p "$M"
cd "$WT" || exit 2

# name | package | file | sed expression | test regexp
run() {
  name="$1"; pkg="$2"; file="$3"; expr="$4"; test="$5"
  echo "$name" | grep -Eq "$ONLY" || return
  cp "$WT/$file" "$M/$name.go"
  sed -i '' -e "$expr" "$M/$name.go"
  if cmp -s "$WT/$file" "$M/$name.go"; then echo "CONTROL $name: FAULT NOT APPLIED"; return; fi
  printf '{"Replace":{"%s":"%s"}}' "$WT/$file" "$M/$name.go" > "$M/$name.json"
  out=$(go test -overlay="$M/$name.json" "$pkg" -run "$test" -count=1 -timeout 150s 2>&1)
  code=$?
  failed=$(echo "$out" | grep -E "^--- FAIL" | sed 's/ (.*//' | tr '\n' ' ')
  if [ $code -eq 0 ]; then echo "CONTROL $name: UNDETECTED (tests passed) <- $test"
  elif [ -z "$failed" ]; then echo "CONTROL $name: NOT A DETECTION (no test failed: the build or the setup did)"
  else echo "CONTROL $name: RED  $failed"; fi
}

# The file that a TypeScript control changed in place is put back when the script ends, however it ends (the restore is
# registered BEFORE the file is changed); a failure to put it back is said.
RESTORE_FILE=""
RESTORE_COPY=""
restore() {
  if [ -n "$RESTORE_FILE" ]; then
    cp "$RESTORE_COPY" "$RESTORE_FILE" || echo "THE FILE $RESTORE_FILE WAS NOT RESTORED: its copy is $RESTORE_COPY"
    RESTORE_FILE=""
  fi
}
trap restore EXIT INT TERM

# name | file | sed expression | vitest files
runts() {
  name="$1"; file="$2"; expr="$3"; tests="$4"
  echo "$name" | grep -Eq "$ONLY" || return
  cp "$WT/$file" "$M/$name.orig"
  RESTORE_FILE="$WT/$file"; RESTORE_COPY="$M/$name.orig"
  sed -i '' -e "$expr" "$WT/$file"
  if cmp -s "$WT/$file" "$M/$name.orig"; then echo "CONTROL $name: FAULT NOT APPLIED"; restore; return; fi
  out=$(CI= pnpm exec vitest run $tests 2>&1)
  code=$?
  restore
  cmp -s "$WT/$file" "$M/$name.orig" || echo "CONTROL $name: THE FILE WAS NOT RESTORED"
  failed=$(echo "$out" | grep -E "^ *(FAIL|×)" | head -3 | tr '\n' ' ')
  if [ $code -eq 0 ]; then echo "CONTROL $name: UNDETECTED (tests passed) <- $tests"
  elif [ -z "$failed" ]; then echo "CONTROL $name: NOT A DETECTION (vitest failed before a test did: the setup)"
  else echo "CONTROL $name: RED  $failed"; fi
}
