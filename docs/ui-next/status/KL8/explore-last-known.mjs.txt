// Scratch (not committed): the landing of a host that restarted on a home that kept the connection's last answer: what does the row say before the connection is opened? KL8.
import { chromium } from '/Users/joshuapare/Repos/omniviewdev/worktrees/landing-pin-2/node_modules/playwright/index.mjs';
import { openPluginPage, openWindow, landingIsUp, readLandingDom } from '/Users/joshuapare/Repos/omniviewdev/worktrees/landing-pin-2/scripts/ui-next-pods/parity.mjs';
const base = process.env.BASE;
const browser = await chromium.launch();
const page = await openWindow(browser, base);
await openPluginPage(page);
await landingIsUp(page);
for (const wait of [300, 1500, 4000]) {
  await page.waitForTimeout(wait);
  const dom = await readLandingDom(page);
  const last = await page.evaluate(() => [...document.querySelectorAll('[data-last-known]')].map((el) => ({ text: el.textContent, tip: el.getAttribute('data-ov-tip') })));
  console.log(`+${wait} ms`, JSON.stringify(dom?.rows), 'last known cells:', JSON.stringify(last));
}
await browser.close();
