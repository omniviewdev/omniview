names = 'TestTheOutputOf|TestASessionThatEnded|TestTheRecordOf|TestTheThirtyThird|TestLiveSessionsAreNot|TestUnloadingAPlugin|TestAnAttachRacing|TestAnEndedSessionTakes|TestClosingAnEnded|TestDetachingAnEnded|TestClosingASessionThatJustEnded|TestAFailedClose|TestASessionTheHostClosed|TestAnAttachToAPluginThatKept|TestACommandWhose|TestACallAboutASession'
only = sys.argv[1:] 
env = dict(os.environ, GOWORK='off', GOTOOLCHAIN='local', GOFLAGS='')
survivors = []
for name, path, old, new, count in M:
    tag = name.split()[0]
    if only and tag not in only: continue
    s = src[path]
    assert s.count(old) == count, (name, s.count(old))
    out = '/tmp/sdk4/mut/%s.go' % tag
    open(out, 'w').write(s.replace(old, new))
    ov = '/tmp/sdk4/mut/%s.json' % tag
    json.dump({'Replace': {path: out}}, open(ov, 'w'))
    r = subprocess.run(['go', 'test', '-count=1', '-overlay=' + ov, './backend/pkg/plugin/exec/', '-run', names, '-v'], cwd=root, env=env, capture_output=True, text=True, timeout=600)
    failed = sorted(set(re.findall(r'^(?:\s*)--- FAIL: (\S+)', r.stdout, re.M)))
    top = sorted(set(f.split('/')[0] for f in failed))
    status = 'KILLED' if r.returncode != 0 and top else ('BUILD-ERR' if 'build failed' in r.stdout + r.stderr or '[build failed]' in r.stdout else 'SURVIVED')
    if status != 'KILLED': survivors.append(name)
    print('%-9s %s\n          -> %s' % (status, name, ', '.join(top) if top else (r.stdout + r.stderr)[-300:].strip()))
print('survivors:', survivors)
