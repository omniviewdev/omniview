import subprocess, sys, shutil, os, re
root='/Users/joshuapare/Repos/omniviewdev/worktrees/s1-host-replay-markers'
def run(cwd, cmd):
    r=subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, shell=True)
    out=r.stdout+r.stderr
    m=re.search(r'Tests\s+(?:(\d+) failed \| )?(\d+) passed', out)
    failed=int(m.group(1) or 0) if m else -1
    return r.returncode, failed, out

def mutate(label, path, old, new, cwd, cmd, expect=None):
    full=os.path.join(root,path)
    src=open(full).read()
    assert old in src, (label, old[:60])
    bak='/tmp/s1host/mut/backup.tmp'
    shutil.copyfile(full,bak)
    try:
        open(full,'w').write(src.replace(old,new,1))
        rc,failed,out=run(os.path.join(root,cwd), cmd)
        names=sorted(set(re.findall(r'FAIL\s+\S+ > (.+)', out)))
        status='KILLED' if rc!=0 else 'SURVIVED'
        print(f'{status:9} {label}: {failed} failed')
        for n in names[:4]: print('          -', n[:150])
    finally:
        shutil.copyfile(bak,full)
