import fs from 'node:fs';
const sec = (a, b) => (Date.parse(b) - Date.parse(a)) / 1000;
const q = (xs, p) => { const s = [...xs].sort((a, b) => a - b); const at = (s.length - 1) * p; const lo = Math.floor(at), hi = Math.ceil(at); return s[lo] + (s[hi] - s[lo]) * (at - lo); };
export function stats(file) {
  const d = JSON.parse(fs.readFileSync(file, 'utf8'));
  const jobs = d.jobs.filter((j) => j.conclusion && j.started_at && j.completed_at);
  const shards = jobs.filter((j) => j.name.startsWith('Kit probes shard '));
  const dur = (j) => sec(j.started_at, j.completed_at);
  const stepSecs = (j, re) => j.steps.filter((s) => re.test(s.name) && s.started_at && s.completed_at).reduce((a, s) => a + sec(s.started_at, s.completed_at), 0);
  const run = shards.map((j) => stepSecs(j, /^Run the units/));
  return {
    jobs: d.jobs.length,
    shards: shards.length,
    runner: jobs.reduce((a, j) => a + dur(j), 0),
    shardRunner: shards.reduce((a, j) => a + dur(j), 0),
    slowest: Math.max(...shards.map(dur)),
    units: run.reduce((a, b) => a + b, 0),
    packages: shards.reduce((a, j) => a + stepSecs(j, /^Build the workspace packages/), 0),
    dist: shards.reduce((a, j) => a + stepSecs(j, /^Build ui-next once/), 0),
    init: shards.reduce((a, j) => a + stepSecs(j, /^Initialize containers/), 0),
    created: Math.min(...d.jobs.map((j) => Date.parse(j.created_at ?? j.started_at))),
    firstStart: Math.min(...jobs.map((j) => Date.parse(j.started_at))),
    lastEnd: Math.max(...jobs.map((j) => Date.parse(j.completed_at))),
  };
}
