import { spawn } from 'node:child_process';
import { mkdtempSync, readFileSync, writeFileSync } from 'node:fs';
import { homedir, tmpdir } from 'node:os';
import path from 'node:path';
import { launch, sleep } from '/Users/joshuapare/Repos/omniviewdev/worktrees/l4-log-rate-open/packages/kit/probes/probe-lib.mjs';
const MOCK_DIR = path.join(homedir(), 'Repos/omniviewdev/aws-wails-native-lab/app/OmniviewNativeLab/frontend/mock');
const copy = path.join(mkdtempSync(path.join(tmpdir(), 'l4-mock-')), 'server.cjs');
writeFileSync(copy, readFileSync(path.join(MOCK_DIR, 'server.js'), 'utf8').replace('const root=__dirname', `const root=${JSON.stringify(MOCK_DIR)}`));
const server = spawn(process.execPath, [copy], { env: { ...process.env, PORT: '49482' }, stdio: 'ignore' });
await sleep(1500);
const out = {};
try {
  for (const engine of ['chromium', 'webkit']) {
    const browser = await launch(engine);
    const context = await browser.newContext({ viewport: { width: 1440, height: 900 }, timezoneId: 'UTC', locale: 'en-US' });
    const page = await context.newPage();
    await page.goto('http://127.0.0.1:49482/', { waitUntil: 'load' });
    await sleep(3000);
    await page.waitForFunction(() => typeof Shell !== 'undefined' && Shell.panel?.logs !== undefined);
    await sleep(1500);
    await page.evaluate(() => { window.__t0 = performance.now(); Shell.panel.logs.open('plugin:aws'); });
    const series = [];
    for (let i = 0; i < 40; i++) {
      series.push(await page.evaluate(() => `${Math.round(performance.now() - window.__t0)}ms ` + (document.querySelector('shell-panel-leaf[data-leaf="logstat"]')?.textContent ?? '(no strip)')));
      await sleep(250);
    }
    out[engine] = series;
    await context.close(); await browser.close();
  }
} finally { server.kill(); }
console.log(JSON.stringify(out, null, 1));
