#!/bin/bash
# Negative controls for T3-channel-wire (docs/ui-next/status/T3-channel-wire.md; HOW-WE-WORK 5). Run from the repository root:
#   bash docs/ui-next/status/T3-channel-wire/mutations.sh.txt [regexp of the control names to run]
# Every control is a fault in a SCRATCH copy of one non-test source file, built with `go test -overlay`: the worktree is never edited.
# For each control the tests named for it run twice:
#   half 1: with the fault. They must fail, by an assertion (no panic, no build failure, no timeout of the binary).
#   half 2: with the fault and the assertions the control is named for switched off (guarded() in the tests reads
#           TERMINAL_CHANNEL_CONTROL_OFF). They must pass: nothing else in them catches the fault.
export GOWORK=off GOTOOLCHAIN=local
WT=${ROOT:-$PWD}
ONLY=${ONLY:-${1:-.}}
M=$(mktemp -d)
trap 'rm -rf "$M"' EXIT
cd "$WT" || exit 2
T=backend/pkg/terminal

# control NAME FILE TESTS GUARDS OLD NEW
control() {
  name="$1"; file="$2"; tests="$3"; guards="$4"; old="$5"; new="$6"
  echo "$name" | grep -Eq "$ONLY" || return
  cp "$WT/$file" "$M/$name.go"
  if ! OLD="$old" NEW="$new" F="$M/$name.go" python3 - <<'PY'
import os, sys
p = os.environ["F"]; s = open(p).read(); o = os.environ["OLD"]
if s.count(o) != 1:
    sys.exit(1)
open(p, "w").write(s.replace(o, os.environ["NEW"]))
PY
  then echo "CONTROL $name: FAULT NOT APPLIED (the old text is not in $file exactly once)"; return; fi
  printf '{"Replace":{"%s":"%s"}}' "$WT/$file" "$M/$name.go" > "$M/$name.json"
  out1=$(go test -overlay="$M/$name.json" "./$T" -run "^($tests)\$" -count=1 -timeout 240s 2>&1); code1=$?
  failed=$(echo "$out1" | grep -E '^\s*--- FAIL' | sed -E 's/^\s*--- FAIL: ([A-Za-z0-9_]+).*/\1/' | sort -u | tr '\n' ' ')
  out2=$(TERMINAL_CHANNEL_CONTROL_OFF="$guards" go test -overlay="$M/$name.json" "./$T" -run "^($tests)\$" -count=1 -timeout 240s 2>&1); code2=$?
  if [ $code1 -eq 0 ]; then half1="UNDETECTED"
  elif echo "$out1" | grep -Eq 'panic:|\[build failed\]|\[setup failed\]|test timed out'; then half1="NOT AN ASSERTION (panic, build or timeout)"
  elif ! echo "$out1" | grep -q 'Error Trace\|never saw\|Condition never satisfied\|stream closed before'; then half1="NOT AN ASSERTION"
  else half1="RED ($failed)"; fi
  if [ $code2 -eq 0 ]; then half2="GREEN with $guards off"; else half2="STILL RED with $guards off: $(echo "$out2" | grep -E '^\s*--- FAIL' | head -3 | tr '\n' ' ')"; fi
  echo "CONTROL $name: half 1 $half1 | half 2 $half2"
}
