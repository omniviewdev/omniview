#!/bin/bash
# bundle.sh <binary> <app dir> <id suffix> <worktree>
set -e
A=$2; rm -rf $(dirname $A); mkdir -p $A/Contents/MacOS $A/Contents/Resources
cp $4/build/darwin/icons.icns $A/Contents/Resources/; cp $1 $A/Contents/MacOS/Omniview; cp $4/build/darwin/Info.plist $A/Contents/
/usr/libexec/PlistBuddy -c "Set :CFBundleIdentifier dev.omniview.s2b.$3" $A/Contents/Info.plist
codesign --force --deep --sign - $A 2>&1 | tail -1
