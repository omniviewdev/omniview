#!/bin/bash
# Negative controls for T5-inject (shell integration of zsh and fish: backend/pkg/terminal/integration*.go and the scripts in
# backend/pkg/terminal/integration_scripts). Each fault is put into a SCRATCH copy of one source file (a Go file or an embedded
# script) and built with `go test -overlay`; the worktree is never edited. A control has two halves:
#   1. with every test of the integration, the tests that go red are exactly the ones named for it (a fault that goes red in a
#      test it is not named for, or in none, is reported);
#   2. with the named tests skipped (the assertion removed), the same fault passes: nothing else sees it.
# fish must be runnable for the fish subtests to run (a machine with no fish skips them by name, "NOT RUN", and a control
# whose named tests are only fish's is then reported as undetected). A subtest is named by its test: either half is on the
# whole test.
# Run from anywhere in the worktree:  bash docs/ui-next/status/T5-inject/mutations.sh.txt [control-name]
# A control named "?" prints what goes red (a discovery run) instead of comparing.
export GOWORK=off GOTOOLCHAIN=local
WT=$(git rev-parse --show-toplevel) || exit 2
PKG=./backend/pkg/terminal/
RUN='^(TestAShellStartedWithTheIntegrationPrintsTheMarksWithTheSessionNonceAroundACommand|TestTheIntegrationPrintsNothingOnTheTerminalOutsideItsMarks|TestTheUsersStartupFilesAreByteIdenticalAfterASession|TestAZDOTDIRTheUserSetIsHonouredAndRestored|TestTheNonceIsInNoVariableOfTheShellAndNoChildGetsIt|TestTheNonceIsInNoLogNoErrorAndNoAnswerThatTheHostGives|TestANestedLoginShellDoesNotInheritTheIntegration|TestTheIntegrationKeepsTheUsersPathOrderOnTheLaunchEnvironment|TestNoDescriptorOfTheChannelReachesAProgramOfTheUser|TestAMultiLineCommandIsRecordedWithItsNewlinesAndNotWithoutThem|TestTheFishVersionGateReadsTheMajorVersion|TestAFishUserWhoBoundEnterKeepsTheirBindingAndTheCommandHasNoTrustedLine|TestAFishAbbreviationIsRecordedExpandedAsWhatRan|TestTheFishEscapeOfA4KBLineWithControlBytesIsCorrectAndCheap|TestZshXtraceNeverPrintsTheNonce|TestTheFactSaysWhyTheIntegrationIsOffOrSkipped|TestWithTheSettingOffARealShellStartsWithNothingOfTheIntegration|TestTheSettingIsAskedForEveryTerminal|TestTheIntegrationDirectoryIsOwnerOnlyAndOwnedByTheUser|TestTheIntegrationDirectoryIsRemovedOnEveryEnd|TestAShellThatDiesAtOnceWithTheIntegrationIsStartedOnceMoreWithoutIt|TestAShellThatAlwaysDiesIsStartedTwiceAndNoMore|TestAShellThatDiesAtOnceWithoutTheIntegrationIsNotStartedAgain|TestTwoHundredStartsAndKillsLeaveNoDirectoryNoProcessAndNoDescriptor|TestTheFactIsOnOnlyAfterTheShellSaidItIsReady|TestARealShellWhoseScriptNeverRunsIsUnconfirmedAndNotOn|TestASettingThatCannotBeReadFailsClosed|TestBashIsStartedWithItsOwnRcfileAndWithoutTheLoginFlag|TestBashBefore4_1IsSkippedAndStartedAsBefore|TestABashWhoseVersionCannotBeToldIsSkippedAndNotGuessed|TestTheRealBashProbeSaysItsVersion|TestTheLoginEmulationRunsTheFirstLoginFileOnceAndNoOtherOne|TestTheBashrcThatTheLoginFileSourcesRunsOnce|TestAUserPromptCommandStillSeesTheStatusOfTheCommand|TestAnArrayPromptCommandIsKeptAndSeesTheStatus|TestAUsersDebugTrapAndPromptCommandKeepWorkingNextToTheScript|TestAHistoryThatDoesNotHaveTheLineGivesNoLineAndNeverAWrongOne|TestTheScriptNeverClobbersBASHREMATCHOrTheStatus|TestBashXtraceNeverPrintsTheNonceAndTheStatusesAreRight|TestBashClosesTheChannelBeforeAProgramOfTheUserRuns|TestABashLineWithControlBytesAndSeparatorsIsEscapedAndRecordedAsTyped|TestPwshIsStartedWithNoExitAndTheScriptAsItsCommand|TestPwshStatusIsExitCodeOfANativeCommandAndOneForAFailedCmdlet|TestAHangupDuringThePromptCommandsStillHangsUpTheJobs)$'
M=$(mktemp -d "${TMPDIR:-/tmp}/t5inject-mut.XXXXXX") || { echo "no scratch directory" >&2; exit 2; }
trap 'rm -rf -- "$M"' EXIT
cd "$WT" || exit 2
only="${1:-}"
bad=0

# name | file | sed script | the tests named for it (sorted, separated by spaces) | optional: tests that may also go red
# (racy detections): half 1 passes when the red set is the named set plus any subset of the optional set.
run() {
  name="$1"; file="$2"; expr="$3"; named="$4"; optional="${5:-}"
  [ -n "$only" ] && [ "$only" != "$name" ] && return
  cp "$WT/$file" "$M/$name.src"
  sed -i -e "$expr" "$M/$name.src"
  if cmp -s "$WT/$file" "$M/$name.src"; then echo "CONTROL $name: FAULT NOT APPLIED"; bad=1; return; fi
  printf '{"Replace":{"%s":"%s"}}' "$WT/$file" "$M/$name.src" > "$M/$name.json"

  out=$(go test -overlay="$M/$name.json" -run "$RUN" "$PKG" -timeout 600s 2>&1)
  code=$?
  failed=$(echo "$out" | grep -E '^--- FAIL' | sed -E 's/--- FAIL: ([A-Za-z0-9_]+).*/\1/' | sort | tr '\n' ' ' | sed 's/ $//')
  if [ "$named" = "?" ]; then echo "CONTROL $name: red in [$failed]"; return; fi
  if [ $code -eq 0 ]; then half1="UNDETECTED (every test passed)"; bad=1
  elif [ -z "$failed" ]; then half1="NOT A DETECTION (the build or the setup failed)"; bad=1
  else
    core=$(for t in $failed; do case " $optional " in *" $t "*) ;; *) echo "$t";; esac; done | tr '\n' ' ' | sed 's/ $//')
    if [ "$core" != "$named" ]; then half1="RED in [$failed], named [$named] (optional [$optional])"; bad=1
    else half1="RED in [$failed] = named plus optional"; fi
  fi

  skip="^($(echo "$named $optional" | sed 's/  *$//; s/ /|/g'))\$"
  out2=$(go test -overlay="$M/$name.json" -run "$RUN" -skip "$skip" "$PKG" -timeout 600s 2>&1)
  if [ $? -eq 0 ]; then half2="without them: PASS (undetected)"
  else half2="without them: STILL RED [$(echo "$out2" | grep -E '^--- FAIL' | tr '\n' ' ')]"; bad=1; fi
  echo "CONTROL $name: $half1 | $half2"
}
