#!/bin/bash
# Negative controls for T4-restart, the page half (ui-next/shell/panel/terminal: execFeed.ts, localShell.ts, TerminalSessions.ts, screen.ts, TerminalBody.tsx, state.ts, terminals.ts, terminalChannel.ts).
# Each fault is put into the real file (a backup is kept and restored on exit, even on Ctrl-C), and the test files of the folder are run with vitest. A control has two halves:
#   1. with every test, the tests that go red are exactly the ones named for it;
#   2. with the named tests filtered out (the assertion removed), the same fault passes.
# Run from the worktree:  bash docs/ui-next/status/T4-restart/page-mutations.sh.txt [control-name]
WT=$(git rev-parse --show-toplevel) || exit 2
cd "$WT" || exit 2
P=ui-next/shell/panel/terminal
FILES="$P"
B=$(mktemp -d "${TMPDIR:-/tmp}/t4restart-page.XXXXXX") || exit 2
current=""
restore() { if [ -n "$current" ] && [ -f "$B/backup" ]; then cp "$B/backup" "$current"; fi; }
trap 'restore; rm -rf -- "$B"' EXIT
only="${1:-}"
bad=0
matched=0

# name | file | sed expression | the tests named for it, each as a unique substring of its title, separated by '|'
run() {
  name="$1"; file="$2"; expr="$3"; named="$4"
  [ -n "$only" ] && [ "$only" != "$name" ] && return
  matched=1
  cp "$file" "$B/backup"; current="$file"
  sed -i -E -e "$expr" "$file"
  if cmp -s "$file" "$B/backup"; then echo "CONTROL $name: FAULT NOT APPLIED"; bad=1; restore; current=""; return; fi

  out=$(pnpm exec vitest run $FILES 2>&1)
  failed=$(echo "$out" | grep -E '^ FAIL ' | sed -E 's/^ FAIL +[^>]+> //' | sort -u)
  ok=1
  if [ -z "$failed" ]; then half1="UNDETECTED (every test passed)"; ok=0
  else
    half1="RED in exactly the tests named"
    # every part names a test that went red, and every test that went red is named by a part
    while IFS= read -r part; do
      echo "$failed" | grep -qF -- "$part" || { ok=0; half1="named test did not go red: $part"; }
    done < <(echo "$named" | tr '|' '\n')
    while IFS= read -r line; do
      hit=0
      while IFS= read -r part; do
        case "$line" in *"$part"*) hit=1 ;; esac
      done < <(echo "$named" | tr '|' '\n')
      [ "$hit" -eq 0 ] && { ok=0; half1="RED outside the named: $line"; }
    done < <(echo "$failed")
  fi
  [ "$ok" -eq 0 ] && bad=1

  # `-t` takes a pattern: every test whose title holds none of the named parts.
  alternatives=$(echo "$named" | tr '|' '\n' | sed -E 's/[]\[.*^$()+?{|\\]/\\&/g' | paste -sd'|')
  out2=$(pnpm exec vitest run $FILES -t "^(?!.*($alternatives))" 2>&1)
  code2=$?
  if echo "$out2" | grep -qE '^ FAIL '; then half2="without them: STILL RED"; bad=1
  elif [ "$code2" -ne 0 ]; then half2="without them: the run ended with status $code2 and no FAIL line (not a pass)"; bad=1
  else half2="without them: PASS (undetected)"; fi
  restore; current=""
  echo "CONTROL $name: $half1 | $half2"
}
