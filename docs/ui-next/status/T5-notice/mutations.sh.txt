#!/bin/bash
# Negative controls of the tracked directory (issue #694 step 6). Each control puts ONE fault in a scratch copy of one source file and runs
# the tests with `go test -overlay`, the worktree untouched. A control is good when (1) the run FAILS in the assertion it is named for (the guard of
# cwd_tracked_test.go, `guarded(t, "<name>", ...)`) and (2) the same fault PASSES with that guard switched off
# (TERMINAL_CHANNEL_CONTROL_OFF=<name>): a fault that a guard catches for another reason proves nothing.
#
# usage: bash mutations.sh.txt            (from the worktree root; Go 1.27 on PATH, GOWORK=off GOTOOLCHAIN=local)
set -u
ROOT=$(pwd)
PKG=./backend/pkg/terminal
TESTS='TestTheDirectoryIsTheOne|TestADirectoryNoNonce|TestAProvedDirectory|TestTheEventsLanes|TestATerminalWithoutIntegrationAnswers|TestAProofThat|TestTheShellsOwn'
WORK=$(mktemp -d)
trap 'rm -rf "$WORK"' EXIT

# control name | file | python expression that makes the fault: (old, new)
BAD=0
run() {
  name=$1 file=$2 old=$3 new=$4
  # ONLY="name name" runs those controls alone.
  if [ -n "${ONLY:-}" ] && [[ " $ONLY " != *" $name "* ]]; then return; fi
  cp "$ROOT/$file" "$WORK/mut.go"
  python3 - "$WORK/mut.go" "$old" "$new" <<'PY'
import sys
p, old, new = sys.argv[1:4]
s = open(p).read()
assert old in s, "the fault does not apply: " + old
open(p, 'w').write(s.replace(old, new, 1))
PY
  [ $? -eq 0 ] || { echo "CONTROL $name: DOES NOT APPLY"; BAD=1; return; }
  printf '{"Replace":{"%s":"%s"}}' "$ROOT/$file" "$WORK/mut.go" > "$WORK/overlay.json"
  # With the guard: the run must FAIL, in a test (not in the build). Guard off: the same fault must PASS.
  caught=$(go test -overlay="$WORK/overlay.json" "$PKG" -run "$TESTS" 2>&1); s1=$?
  off=$(TERMINAL_CHANNEL_CONTROL_OFF=$name go test -overlay="$WORK/overlay.json" "$PKG" -run "$TESTS" 2>&1); s2=$?
  if [ $s1 -ne 0 ] && echo "$caught" | grep -q '^--- FAIL'; then c1=RED; else c1=GREEN-OR-BROKEN; fi
  if [ $s2 -eq 0 ]; then c2=GREEN; else c2=RED-OR-BROKEN; fi
  echo "CONTROL $name: with the guard $c1 (want RED), guard off $c2 (want GREEN)"
  if [ "$c1" != RED ] || [ "$c2" != GREEN ]; then BAD=1; fi
}

run no-tracking,tracked,follows,logical-spelling,events-tracked backend/pkg/terminal/manager.go 'tracked, ok := s.trackedDirectory()' 'tracked, ok := s.trackedDirectory(); ok = false'
run stale-proof,events-stale-proof backend/pkg/terminal/manager.go 'os.SameFile(trackedInfo, pulledInfo) {' 'os.SameFile(trackedInfo, pulledInfo) || true {'
run logical-spelling backend/pkg/terminal/manager.go 'os.SameFile(trackedInfo, pulledInfo) {
		return tracked, ""' 'os.SameFile(trackedInfo, pulledInfo) {
		return pulled, ""'
exit $BAD
