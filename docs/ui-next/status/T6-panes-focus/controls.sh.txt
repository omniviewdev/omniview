#!/bin/bash
# Negative controls for the pane chords, the focus between panes and the panes after a reload (#697 step 6, the last part). Run from the root of the worktree:
#   bash docs/ui-next/status/T6-panes-focus/controls.sh.txt
# Each fault goes into the real file, which is put back after each control (in a `finally`, and on a signal, from a copy kept in memory): the page tests render the shell's own tree, so a scratch copy of the panel would not be the code
# they draw. A control is RED when the tests named for it fail with the fault in. The same control is then run with the named tests skipped (the assertion removed): what is still red is caught by another test as well. The second half
# is ENFORCED: SOLE lists the controls whose named tests alone guard them, and a control whose second run disagrees counts as not caught. A baseline run of the untouched files must be green first. Exit code: 0 when every control was
# caught by the tests named for it and agreed with SOLE, 1 otherwise. The faults of the page in a real engine (a pane under 80 px after a drag, a sash that is not a Tab stop, a context for each showing) are the probe's own table:
# `terminal-controls-panes` (terminal-panes.mjs: panes-chords-lost, panes-chords-sent-to-shell, panes-reload-forgets-layout, panes-gone-kept, panes-ended-closed). Do not run it while someone else edits these files.
set -u
ROOT="$(pwd)"
OUT="$(mktemp -d)"
trap 'rm -rf "$OUT"' EXIT
export ROOT OUT
python3 - <<'PY'
import atexit, json, os, re, signal, subprocess, sys

root, out = os.environ['ROOT'], os.environ['OUT']
SHELL = 'ui-next/shell'
# The tests of this part: the chords, the model's panes, the page tests of the panes, the registry's, the keys and the session.
FILES = [
  f'{SHELL}/core/chords.test.ts', f'{SHELL}/core/panel/panes.test.ts', f'{SHELL}/core/panel/panes.fuzz.test.ts', f'{SHELL}/panel/panes/Panes.test.tsx', f'{SHELL}/panel/panes/PaneEmulators.test.tsx',
  f'{SHELL}/panel/terminal/TerminalSplit.test.tsx', f'{SHELL}/panel/terminal/terminals.test.ts', f'{SHELL}/panel/terminal/keys.test.ts', f'{SHELL}/session/session.test.ts',
]

# name | file (under ui-next/shell) | exact text | replacement | regexp of "file > test title" that must be among the failures
CONTROLS = [
  ('pane-chord-works-outside-the-panel', 'core/chords.ts', 'if (!inPanel || event.defaultPrevented === true || event.isComposing === true || event.keyCode === 229) {', 'if (event.defaultPrevented === true || event.isComposing === true || event.keyCode === 229) {', r'chords\.test\.ts.*the pane chords are the panel'),
  ('pane-chord-takes-alt-on-linux', 'core/chords.ts', '&& !event.metaKey && !event.altKey && event.getModifierState?.(\'AltGraph\') !== true;\n  if (!held) {', '&& !event.metaKey && event.getModifierState?.(\'AltGraph\') !== true;\n  if (!held) {', r'chords\.test\.ts.*not the other platform'),
  ('pane-chord-up-down-on-linux', 'core/chords.ts', "return platform === 'mac' ? { action: 'up' } : undefined;", "return { action: 'up' };", r'chords\.test\.ts.*Windows and Linux'),
  ('next-does-not-wrap', 'core/panel/panes.ts', 'return layout.panes[(at + 1) % count].id;', 'return layout.panes[Math.min(at + 1, count - 1)].id;', r'panes\.test\.ts.*goes to the next and the previous'),
  ('left-wraps', 'core/panel/panes.ts', 'return layout.panes[at + (step === \'left\' || step === \'up\' ? -1 : 1)]?.id;', 'return layout.panes[(at + count + (step === \'left\' || step === \'up\' ? -1 : 1)) % count].id;', r'panes\.test\.ts.*stops at the ends'),
  ('up-moves-in-a-row', 'core/panel/panes.ts', "const along = layout.axis === 'row' ? step === 'left' || step === 'right' : true;", 'const along = true;', r'panes\.test\.ts.*goes left and right in a row'),
  ('a-leaving-pane-keeps-its-share', 'core/panel/sessions.ts', 'ratios: Object.freeze(others.map((ratio) => ratio / total)),', 'ratios: Object.freeze(others.map((ratio) => ratio)),', r'panes\.test\.ts.*a pane that leaves'),
  ('a-session-in-no-pane-may-leave', 'core/panel/panes.ts', 'group?.panes.some((pane) => pane.session === session) !== true', 'group === undefined', r'panes\.test\.ts.*refuses a session that is in no pane'),
  ('the-chord-does-not-make-the-pane-active', 'panel/focus.ts', '  store.focusPane(shown.tool, target);\n  focusPaneBody(pane.session);', '  focusPaneBody(pane.session);', r'Panes\.test\.tsx.*(moves the keyboard to the pane on the left|makes the pane the active one even when its body)'),
  ('the-chord-does-not-move-the-keyboard', 'panel/focus.ts', '  store.focusPane(shown.tool, target);\n  focusPaneBody(pane.session);', '  store.focusPane(shown.tool, target);', r'Panes\.test\.tsx.*moves the keyboard to the pane on the left'),
  ('the-chord-is-not-cancelled', 'panel/usePanelChords.ts', 'if (moveToPane(store, pane.action)) {\n          event.preventDefault();', 'if (moveToPane(store, pane.action)) {\n          void 0;', r'Panes\.test\.tsx.*is cancelled when it moved the keyboard'),
  ('the-chord-works-outside-the-panel', 'panel/usePanelChords.ts', 'paneChordOf(event, platform, inPanel(event.target))', 'paneChordOf(event, platform, true)', r'Panes\.test\.tsx.*does nothing outside the panel'),
  ('the-pane-is-not-in-the-name', 'panel/terminal/TerminalBody.tsx', "${seat === null || seat.count < 2 ? '' : `pane ${String(seat.index + 1)} of ${String(seat.count)}, `}", "${''}", r'TerminalSplit\.test\.tsx.*says which pane it is in'),
  ('a-gone-terminal-keeps-its-pane', 'panel/terminal/terminals.ts', '    leaveGonePanes();\n  };', '  };', r'terminals\.test\.ts.*leaves the pane of a terminal that is gone'),
  ('every-terminal-leaves-its-pane', 'panel/terminal/terminals.ts', 'if (sessions.feedOf(pane.session) === undefined) {', 'if (true as boolean) {', r'terminals\.test\.ts.*brings the group back as it was'),
  ('a-silent-host-closes-the-panes', 'panel/terminal/terminals.ts', "console.warn('[panel] the terminals left for this window could not be found again', error);\n        return;", "console.warn('[panel] the terminals left for this window could not be found again', error);\n        leaveGonePanes();\n        return;", r'terminals\.test\.ts.*keeps the panes of a host that could not answer'),
]

# The controls whose fault the named tests alone catch: with those tests skipped the copy passes. Every other control stays red with them skipped (another test catches it as well).
SOLE = {
  'pane-chord-takes-alt-on-linux', 'pane-chord-up-down-on-linux', 'up-moves-in-a-row', 'a-session-in-no-pane-may-leave', 'the-chord-is-not-cancelled', 'the-pane-is-not-in-the-name',
  'a-silent-host-closes-the-panes', 'the-chord-works-outside-the-panel', 'the-chord-does-not-make-the-pane-active',
}

TITLES = {}  # 'file > full name' -> the test's own title: vitest matches a name pattern against the title and its describes joined in a way a full name is not, so a skip uses the title alone

def tests(skip=None):
    """Runs the tests of the copy; returns (exit code, set of 'file > title' that failed)."""
    report = os.path.join(out, 'report.json')
    if os.path.exists(report):
        os.remove(report)
    cmd = ['./node_modules/.bin/vitest', 'run', *FILES, '--reporter=json', f'--outputFile={report}']
    if skip:
        escaped = [re.sub(r'[.*+?^${}()|\[\]\\/]', lambda m: '\\' + m.group(0), s) for s in skip]
        cmd += ['--testNamePattern=^(?!.*(?:' + '|'.join(escaped) + ')).*$']
    code = subprocess.run(cmd, cwd=root, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL).returncode
    failed = set()
    try:
        data = json.load(open(report))
        for file in data['testResults']:
            name = os.path.basename(file['name'])
            for t in file['assertionResults']:
                if t['status'] == 'failed':
                    failed.add(f"{name} > {t['fullName']}")
                    TITLES[f"{name} > {t['fullName']}"] = t['title']
            if file['status'] == 'failed' and not any(t['status'] == 'failed' for t in file['assertionResults']):
                failed.add(f'{name} > (the file did not run)')
    except Exception as error:  # noqa: BLE001
        failed.add(f'(no report: {error})')
    return code, failed

ORIGINAL = {}  # path -> its text before a fault was put in

def restore():
    for path, text in ORIGINAL.items():
        open(path, 'w').write(text)
    ORIGINAL.clear()

def plant(path, old, new):
    """Puts the fault in; False when the text it replaces is not there exactly once."""
    text = open(path).read()
    if text.count(old) != 1:
        return False
    ORIGINAL[path] = text
    open(path, 'w').write(text.replace(old, new))
    return True

atexit.register(restore)
for signum in (signal.SIGINT, signal.SIGTERM, signal.SIGHUP):
    signal.signal(signum, lambda *_: sys.exit(130))

code, failed = tests()
print(f'BASELINE the untouched files: exit {code}, {len(failed)} failed')
if code != 0:
    print('  baseline is not green: ' + '; '.join(sorted(failed))[:600])
    sys.exit(2)

bad = 0
for name, file, old, new, expect in CONTROLS:
    path = os.path.join(root, SHELL, file)
    if not plant(path, old, new):
        print(f'CONTROL {name}: FAULT NOT APPLIED ({file}: the text does not occur exactly once)')
        bad += 1
        continue
    code, failed = tests()
    named = sorted(t for t in failed if re.search(expect, t))
    if code == 0 or not failed:
        restore()
        print(f'CONTROL {name}: UNDETECTED (the tests passed)')
        bad += 1
        continue
    status = 'RED' if named else 'RED, BUT NOT BY THE TEST NAMED FOR IT'
    if not named:
        bad += 1
    print(f'CONTROL {name}: {status}: {len(failed)} test(s) fail, {len(named)} of them named for it')
    for t in (named or sorted(failed))[:3]:
        print(f'    {t}')
    if named:
        titles = [TITLES[t] for t in named]
        code2, rest = tests(skip=titles)
        alone = code2 == 0 and not rest
        print(f'    with the named tests skipped: exit {code2}, {len(rest)} other test(s) still fail' + (f' (e.g. {sorted(rest)[0]})' if rest else ' (the fault goes undetected: those tests alone guard it)'))
        if alone != (name in SOLE):
            print(f'    MISMATCH: expected {"the named tests alone to guard it" if name in SOLE else "another test to catch it as well"}')
            bad += 1
    restore()

print('ALL CONTROLS CAUGHT BY THEIR TEST' if bad == 0 else f'{bad} CONTROL(S) NOT CAUGHT')
sys.exit(1 if bad else 0)
PY
code=$?
rm -rf "$OUT"
exit $code
