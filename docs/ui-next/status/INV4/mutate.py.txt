#!/usr/bin/env python3
# INV4 mutation runner. It applies one fault to a scratch copy of the plugin's package (never to a worktree), runs the tests named for the faults and says whether the fault is CAUGHT.
# The rule (docs/ui-next/AGENTS.md 28: a negative control must trip the assertion it is named for, and the same fault must pass undetected when that assertion is removed):
#   1. The baseline runs first, on the unmutated copy: every selected test must pass, twice. A red baseline stops the run and nothing is counted (a test that fails WITHOUT the fault cannot detect it).
#   2. A fault is CAUGHT only by tests that pass at the baseline and fail with the fault, and that fail again when only they are run again with it (a test that fails once on a fault and passes on the same fault is noise).
#   3. A fault that does not apply, breaks the build, passes every test, or is seen only by tests that EXPECT does not name for it is NOT caught, and the run exits 1.
#   4. The converse (--converse) must first see the fault trip an assertion of a test named for it. It then removes exactly the one-line assertions that tripped, round by round, and the run with them removed must pass.
# Usage (INV4_MUT_BASE is a copy of go.mod, go.sum, pkg/ and ui/src/kinds/ of the plugin under test; INV4_WORK holds the scratch copies; INV4_MUT_OUT, if set, is a JSON file for the results):
#   python3 mutate.py                 every fault          python3 mutate.py M35 M36        these faults          python3 mutate.py --converse M1 M2        the converse of these          python3 mutate.py --baseline

ENV = dict(os.environ, GOWORK='off', GOTOOLCHAIN='local', GOFLAGS='')
def short(name): return name.replace('TestNative', '', 1)
def ident(key): return key.split(' ')[0]
def copy_base(prefix):
    os.makedirs(SCRATCH, exist_ok=True)
    d = tempfile.mkdtemp(prefix=prefix, dir=SCRATCH)
    shutil.copytree(BASE, d, dirs_exist_ok=True)
    return d
def apply(d, spec):
    """Applies the fault to the copy. None when it applied, else why it did not."""
    for path, old, new in (spec if isinstance(spec, list) else [spec]):
        s = open(os.path.join(d, path)).read()
        if old not in s: return f'{path}: the text to replace is not there'
        open(os.path.join(d, path), 'w').write(s.replace(old, new, 1))
    return None
def go_test(d, pattern, count=1):
    p = subprocess.run(['go', 'test', './pkg/plugin/resource/', f'-count={count}', '-timeout', '8m', '-v', '-run', pattern], cwd=d, env=ENV, capture_output=True, text=True, errors='replace', timeout=1200)
    return p.returncode, p.stdout + p.stderr
def built(out): return '[build failed]' not in out and '[setup failed]' not in out
def verdicts(out):
    """The top-level tests that passed and that failed (a subtest is indented and belongs to its test)."""
    return set(re.findall(r'^--- PASS: (\S+)', out, re.M)), set(re.findall(r'^--- FAIL: (\S+)', out, re.M))
def only(names): return '^(' + '|'.join(re.escape(n) for n in sorted(names)) + ')$'
def baseline():
    """The unmutated copy: every selected test must pass, twice. Returns the tests that passed and the problems (none when the baseline is green)."""
    d = copy_base('base-')
    try:
        rc, out = go_test(d, RUN, count=2)
    finally:
        shutil.rmtree(d, ignore_errors=True)
    if not built(out): return set(), ['the baseline does not build']
    passed, failed = verdicts(out)
    problems = []
    if failed: problems.append('the baseline is red: ' + ', '.join(sorted(short(t) for t in failed)))
    elif rc != 0: problems.append(f'the baseline exits {rc}')
    if not passed: problems.append('the baseline ran no test')
    return passed, problems
def run(key, spec, passed_at_baseline):
    """-> (status, the tests that caught the fault, the tests that failed and were dropped as noise)."""
    d = copy_base('mut-')
    try:
        why = apply(d, spec)
        if why: return 'DID NOT APPLY: ' + why, [], []
        rc, out = go_test(d, RUN)
        if not built(out): return 'BUILD ERROR: ' + '; '.join(re.findall(r'^\S+\.go:\d+:\d+: .*$', out, re.M))[:200], [], []
        _, failed = verdicts(out)
        if not failed: return ('NOT CAUGHT: every selected test passes' if rc == 0 else f'ERROR: exit {rc} and no test is reported as failed (a timeout or a crash)'), [], []
        # The noise filter: a test counts only if it passed at the baseline, fails again when only it is run again with the fault, and passes on a pristine copy (twice).
        stay = {t for t in failed if t in passed_at_baseline}
        dropped = sorted(failed - stay)
        if stay:
            _, again = verdicts(go_test(d, only(stay))[1])
            dropped += sorted(stay - again)
            stay &= again
        if stay:
            b = copy_base('pristine-')
            try:
                _, fails_without = verdicts(go_test(b, only(stay), count=2)[1])
            finally:
                shutil.rmtree(b, ignore_errors=True)
            dropped += sorted(stay & fails_without)
            stay -= fails_without
        if not stay: return 'NOT CAUGHT: the tests that failed are noise (they fail without the fault, or do not fail again with it)', [], dropped
        return 'CAUGHT', sorted(stay), dropped
    finally:
        shutil.rmtree(d, ignore_errors=True)
def converse(key, rounds=6):
    """The converse run: the same fault with exactly the assertions that tripped removed. The fault must first trip an assertion of a test named for it; the run with them removed must then pass. -> (status, the assertions removed)."""
    d = copy_base('conv-')
    try:
        why = apply(d, MUTANTS[key])
        if why: return 'DID NOT APPLY: ' + why, []
        named = set(EXPECT.get(ident(key), []))
        removed = []
        for i in range(rounds + 1):
            rc, out = go_test(d, RUN)
            if not built(out): return 'BUILD ERROR', removed
            _, failed = verdicts(out)
            if i == 0:
                if rc == 0 and not failed: return 'NOT CAUGHT: the fault passes every test, so there is no assertion to remove', []
                if named and not ({short(t) for t in failed} & named): return 'the first failure is in no test named for the fault: ' + ', '.join(sorted(short(t) for t in failed))[:160], []
            elif rc == 0:
                return f'ok (passes with the {len(removed)} assertion(s) that tripped removed)', removed
            trips = sorted({(m.group(1), int(m.group(2))) for m in re.finditer(r'Error Trace:\s+(?:\S*/)?(\S+_test\.go):(\d+)', out)})
            if not trips: return 'a failure that is not a testify assertion: ' + '; '.join(sorted(failed))[:160], removed
            for f, ln in trips:
                fp = os.path.join(d, R + f)
                lines = open(fp).read().split('\n')
                l = lines[ln - 1]
                m = re.match(r'^(\s*)(?:require|assert)\.\w+\(\s*t,\s*(.*)\)\s*$', l)
                if not m: return f'cannot remove {f}:{ln} (not a one-line assertion): {l.strip()[:80]}', removed
                # the arguments stay evaluated (no unused-variable build error); only the check goes
                lines[ln - 1] = m.group(1) + '_ = []any{' + m.group(2) + '} // assertion removed'
                open(fp, 'w').write('\n'.join(lines))
                removed.append(f'{f}:{ln}')
        return f'still failing after {rounds} rounds', removed
    finally:
        shutil.rmtree(d, ignore_errors=True)
def pick(n): return next(k for k in MUTANTS if k.startswith(n + ' ') or k == n)
if __name__ == '__main__':
    args = sys.argv[1:]
    passed, problems = baseline()
    if problems:
        print('BASELINE: ' + '; '.join(problems) + '. Nothing is counted.', flush=True)
        sys.exit(2)
    print(f'BASELINE: green, {len(passed)} tests passed, twice', flush=True)
    if args[:1] == ['--baseline']: sys.exit(0)
    bad, results = [], {}
    if args[:1] == ['--converse']:
        for n in args[1:]:
            key = pick(n)
            status, removed = converse(key)
            results[ident(key)] = {'fault': key, 'status': status, 'removed': removed}
            print(f'{key} CONVERSE: {status}; assertions removed ({len(removed)}): ' + ', '.join(removed), flush=True)
            if not status.startswith('ok'): bad.append(key)
    else:
        for n in sorted((a.split(' ')[0] for a in (args or list(MUTANTS))), key=lambda m: int(m[1:])):
            key = pick(n)
            status, caught, dropped = run(key, MUTANTS[key], passed)
            named = set(EXPECT.get(ident(key), []))
            if status == 'CAUGHT' and named and not ({short(t) for t in caught} & named): status = 'CAUGHT, BUT NOT BY A TEST NAMED FOR IT'
            results[ident(key)] = {'fault': key, 'status': status, 'caught_by': [short(t) for t in caught], 'dropped_as_noise': [short(t) for t in dropped], 'named': sorted(named)}
            print(f'{key}: {status}; failing tests ({len(caught)}): ' + ', '.join(short(t) for t in caught) + (f'; dropped as noise: ' + ', '.join(short(t) for t in dropped) if dropped else ''), flush=True)
            if status != 'CAUGHT': bad.append(key)
    if os.environ.get('INV4_MUT_OUT'):
        with open(os.environ['INV4_MUT_OUT'], 'w') as f: json.dump(results, f, indent=1)
    if bad:
        print('NOT CAUGHT, or not by a test named for the fault, or its converse did not pass: ' + '; '.join(bad), flush=True)
        sys.exit(1)
